def obs():return book.observation()
def prior():return (path.read_bytes(),path.stat().st_ctime_ns)
def output(stage,**kw):print(json.dumps(dict(stage=stage,case=case,**kw)),flush=True)
def update(state):
 before=obs()['spent'];p=book.reserve_ledger_operation('update-record',ledger,OperationLimits(syscalls=1024,work=262144,io=16385));after=obs()['spent']

 return p,book.record(state,{'test_only':True,'review':case,'unicode':'\u2603\U0001f642','escaped':'\n\u0001'},p)
def write(p,r):

 output('committed-readback',result=result,full_hex=actual.hex(),ctime_ns=path.stat().st_ctime_ns,account=obs());return result
def refuse(p,r,code):
 before=prior();account=obs();got=None
 try:ledger.write_json(r,p)
 except Refusal as e:got=e.code

 assert got==code and before==after
 assert account['spent']==obs()['spent'] and account['ledger_commits']==obs()['ledger_commits']
 assert obs()['fault'] is not None
 try:book.reserve_phase('audit',issuer,dict(work=1))
 except Refusal:pass
 else:raise AssertionError('fault did not stop new authority')
try:
 parent=ParentLease.acquire(book.parent_path,book.reserve_ledger_operation('acquire-parent',book.parent_path,OperationLimits(fds=4)))
 ledger=OwnedLedger.create(parent,book.basename,book.reserve_ledger_operation('create-ledger',(parent,book.basename),OperationLimits(fds=1)));path=Path(book.parent_path)/book.basename
 if case in ('older-before','older-after'):

  if case=='older-after':write(p2,r2);assert obs()['ledger_commits']==1
  else:assert prior()[0]==b'' and obs()['ledger_commits']==0

  p,r=update('COMPLETE');assert p.record_sequence==3;r['sequence']=2 if case=='duplicate-record' else 1

  refuse(p,r,'InvalidPermit')
 elif case=='phase-interface':

   if phase=='issuer':continue
   reservation=book.reserve_phase(phase,owner,dict(work=103));reservation.check(owner);reservation.charge(owner,work=2);reservation.close(owner)
   assert not hasattr(reservation,'observation')

  p,r=update('COMPLETE');write(p,r);output('phase-interface-fullspend',account=obs())
 else:raise AssertionError(case)
finally:

  if resource is not None:
   resource.close(book.reserve_ledger_operation('cleanup-close',resource,resource.close_requirements()))

 else:raise AssertionError('finished book accepted authority')
