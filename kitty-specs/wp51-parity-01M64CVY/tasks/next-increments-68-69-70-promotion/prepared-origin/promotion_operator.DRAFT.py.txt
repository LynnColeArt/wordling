def create(repo,relative_path,raw,mode=0o644):
    target=repo/relative(relative_path);target.parent.mkdir(parents=True,exist_ok=True)
    fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
    try:
        os.fchmod(fd,mode)
        view=memoryview(raw)
        while view:
            n=os.write(fd,view)
            if n<=0:raise RuntimeError('zero-progress publication')
            view=view[n:]
        os.fsync(fd)
    finally:os.close(fd)
    assert regular(target)==raw

def seal_publication(repo,root):
    directory=repo/root;members=[]
    for p in sorted(directory.rglob('*')):
        if p.is_file():
            raw=regular(p);members.append(dict(path=str(p.relative_to(directory)),bytes=len(raw),sha256=sha(raw),mode=oct(stat.S_IMODE(p.stat().st_mode))))
    raw=(json.dumps(dict(scope='TRANSFORMED literal publication, all original seals included as historical payload; excludes ONLY publication-seal.json',files=members),indent=2)+'\n').encode()
    create(repo,root+'/publication-seal.json',raw)
