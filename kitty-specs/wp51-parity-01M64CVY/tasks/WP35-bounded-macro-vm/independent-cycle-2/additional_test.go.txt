type reviewerCountingSink struct{ count int;bad bool;source [32]byte }
func(s *reviewerCountingSink)Capabilities()mac.Policy{return fullPolicy()}
func(s *reviewerCountingSink)Apply(_ context.Context,e mac.Effect)(mac.EffectResult,error){if e.Kind!="literal"||len(e.Words)!=1||e.Words[0].Value!='r'||e.Words[0].Span.Length!=0||e.Source!=s.source{s.bad=true};s.count++;e.Words[0].Value='Z';e.Words[0].Span=wpm.Span{};return mac.EffectResult{},nil}
func TestReviewerFullLongValueDeliveryHasIndependentWords(t *testing.T){
 raw:=native(t,uint16(0xfc13),"I~"+strings.Repeat("r",4096)+"~~",uint16(0xfc34),"I~",uint16(0xfc0f));im:=image(t,raw);l:=mac.DefaultLimits();l.SliceSteps=1;s:=&reviewerCountingSink{source:im.SHA256()};m:=newMachine(t,program(t,raw,l),s,nil,l);cost:=uint64(im.Inspect().Body.Length/2);for m.State().Steps<cost{m.Step(context.Background(),1)};goruntime.GC();var maximum,total uint64;var r mac.Result
 for n:=0;n<20000;n++{var a,b goruntime.MemStats;goruntime.ReadMemStats(&a);before:=m.State();r=m.Step(context.Background(),1);goruntime.ReadMemStats(&b);d:=b.TotalAlloc-a.TotalAlloc;if d>maximum{maximum=d};total+=d;if r.Steps-before.Steps>1{t.Fatal("quota")};if r.State!="running"{break}}
 t.Logf("4096 delivered words; max-Step-allocation=%d total=%d steps=%d",maximum,total,r.Steps);if r.State!="complete"||s.count!=4096||s.bad||r.EmittedWords!=4096||maximum>64<<10||!bytes.Equal(im.Export(),raw){t.Fatal(r,s.count,s.bad,maximum)}
}
