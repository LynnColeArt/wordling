def sha(raw):return hashlib.sha256(raw).hexdigest()
def regular(path):

def relative(value):

def assert_parents(path,stop):

def create(rel,raw,mode=0o644):

    fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
    try:
        os.fchmod(fd,mode);view=memoryview(raw)
        while view:
            n=os.write(fd,view)
            if n<=0:raise RuntimeError('zero progress')
            view=view[n:]
        os.fsync(fd)
    finally:os.close(fd)
    if regular(target)!=raw:raise RuntimeError('readback mismatch')
