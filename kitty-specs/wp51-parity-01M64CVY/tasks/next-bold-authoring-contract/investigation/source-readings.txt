package document

import (
	"bytes"
)

// Mutation qualification policies, NOT original format limits. Read-only Parse
// retains WP02's 16MiB and WP15's MaxTokens; admission is deliberately narrower.
const (
	MaxMutationBytes  = 65536
	MaxMutationTokens = 16384
	MaxBoldPairs      = 2
)

const (
	RevisionMismatch    Category = "RevisionMismatch"
	MutationLimit       Category = "MutationLimit"
	UnsupportedPrefix   Category = "UnsupportedPrefix"
	UnsupportedResource Category = "UnsupportedResource"
	UnsupportedMetadata Category = "UnsupportedMetadata"
	UnsupportedBody     Category = "UnsupportedBody"
	InvalidASCII        Category = "InvalidASCII"
)

// Revision is an opaque comparable identity, not a content hash. Even separately
// parsed equal bytes have distinct revisions. Its zero value identifies no image.
type Revision struct{ state *revisionState }

func (d *Document) Revision() Revision {
	if !d.valid() {
		return Revision{}
	}
	return Revision{d.state}
}
func (c Cursor) Revision() Revision {
	if !c.state.valid() {
		return Revision{}
	}
	return Revision{c.state}
}

// OriginalExport copies the immutable initially loaded image across all edits.
// New returns explicit NoLoadedOriginal for its entire lifetime. Export instead
// returns this revision's native candidate. Neither performs IO.
func (d *Document) OriginalExport() ([]byte, error) {
	if !d.valid() {
		return nil, failure(InvalidDocument, -1, noSpan)
	}
	if d.state.birth.kind == newBirth {
		return nil, failure(NoLoadedOriginal, -1, noSpan)
	}
	return d.state.birth.originalImage.Export(), nil
}

// EditResult preserves the receiver and incoming cursor on every refusal. No
// partial revision is published. Changed describes a native body change; a
// same-byte typeover only moves its cursor and does not invent metadata dirtiness.
type EditResult struct {
	Document *Document
	Cursor   Cursor
	Changed  bool
	receipt  *asciiEditReceipt
}

// The receipt is completed before publication and never points at exported
// mutable handles. History can validate actual ASCII transitions without trusting
// fabricated/rebound public result fields or retaining document ancestry chains.
type asciiEditReceipt struct {
	before, after         *revisionState
	cursor                Cursor
	changed               bool
	replaced, inputLength int
}

// CheckMutation checks the full reviewed WP23 prefix/resource/paired-body schema.
// Native parsing alone, a matching prefix size or a fixture digest is insufficient.
func (d *Document) CheckMutation() error { _, e := d.mutationAdmission(); return e }

// InsertASCII receives explicit insert semantics; session mode/key delivery is
// outside this package. Input is WP printable ASCII identity, not Unicode text.
func (d *Document) InsertASCII(c Cursor, text []byte) (EditResult, error) {
	return d.editASCII(c, text, false)
}

// TypeoverASCII is repeated original character typeover: replace the next ASCII
// token, insert before a code, append at EOF; advance one resulting token gap per
// character. Encountering a code keeps subsequent characters before that code.
func (d *Document) TypeoverASCII(c Cursor, text []byte) (EditResult, error) {
	return d.editASCII(c, text, true)
}

func (d *Document) editASCII(c Cursor, text []byte, over bool) (EditResult, error) {
	unchanged := EditResult{Document: d, Cursor: c}
	if !d.valid() {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if c.state != d.state {
		return unchanged, failure(RevisionMismatch, c.gap, noSpan)
	}
	// Cursor fields are private, but retain explicit checks before indexing.
	if c.gap < 0 || c.gap > d.TokenCount() {
		return unchanged, failure(InvalidPosition, c.gap, noSpan)
	}
	_, e := d.mutationAdmission()
	if e != nil {
		return unchanged, e
	}
	if len(text) > MaxMutationTokens {
		return unchanged, failure(MutationLimit, c.gap, noSpan)
	}
	for i, b := range text {
		if b < 0x20 || b > 0x7e {
			return unchanged, failure(InvalidASCII, i, noSpan)
		}
	}
	if len(text) == 0 {
		unchanged.receipt = &asciiEditReceipt{before: d.state, after: d.state, cursor: c}
		return unchanged, nil
	}
	raw := d.state.image.Export()
	start := int(d.state.gaps[c.gap])
	end := start
	replaced := 0
	if over {
		for replaced < len(text) && c.gap+replaced < d.TokenCount() {
			span := d.span(c.gap + replaced)
			if identify(raw[span.Offset:span.Offset+span.Length]) != ASCII {
				break
			}
			end += span.Length
			replaced++
		}
	}
	// Subtraction precedes addition, with input text already bounded.
	size := len(raw) - (end - start) + len(text)
	count := d.TokenCount() - replaced + len(text)
	if size > MaxMutationBytes || count > MaxMutationTokens {
		return unchanged, failure(MutationLimit, c.gap, noSpan)
	}
	bodyStart := int(d.state.gaps[0])
	body := make([]byte, 0, size-bodyStart)
	body = append(body, raw[bodyStart:start]...)
	body = append(body, text...)
	body = append(body, raw[end:]...)
	r, e := d.publishBody(c, body, count, c.gap+len(text), false)
	if e == nil {
		r.receipt = &asciiEditReceipt{before: d.state, after: r.Document.state, cursor: r.Cursor, changed: r.Changed, replaced: replaced, inputLength: len(text)}
	}
	return r, e
}

// The sole semantic candidate publisher. Only domain commands call it, after
// whole-token edits. forceFormat is authorized for confirmed blocked-pair repair;
// no-edit/decline/preview paths never call it. It preserves the owned input prefix
// and applies only checked indexed packet6 bit3, never a template or relocation.
func (d *Document) publishBody(c Cursor, body []byte, tokens, gap int, forceFormat bool) (EditResult, error) {
	unchanged := EditResult{Document: d, Cursor: c}
	if !d.valid() {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if c.state != d.state {
		return unchanged, failure(RevisionMismatch, c.gap, noSpan)
	}
	flags, e := d.mutationAdmission()
	if e != nil {
		return unchanged, e
	}
	start := int(d.state.gaps[0])
	if tokens < 0 || tokens > MaxMutationTokens || len(body) > MaxMutationBytes-start {
		return unchanged, failure(MutationLimit, gap, noSpan)
	}
	if gap < 0 || gap > tokens {
		return unchanged, failure(InvalidPosition, gap, noSpan)
	}
	raw := d.state.image.Export()
	changed := !bytes.Equal(raw[start:], body)
	cursor := c
	cursor.gap = gap
	if !changed && (!forceFormat || raw[flags]&8 != 0) {
		return EditResult{Document: d, Cursor: cursor}, nil
	}
	candidate := make([]byte, 0, start+len(body))
	candidate = append(candidate, raw[:start]...)
	candidate = append(candidate, body...)
	candidate[flags] |= 8
	next, e := parseRevision(candidate, d.state.birth)
	if e != nil {
		return unchanged, e
	}
	if next.TokenCount() != tokens {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if e = next.CheckMutation(); e != nil {
		return unchanged, e
	}
	cursor.state = next.state
	return EditResult{Document: next, Cursor: cursor, Changed: changed}, nil
}

func (d *Document) mutationAdmission() (int, error) {
	if !d.valid() {
		return 0, failure(InvalidDocument, -1, noSpan)
	}
	info := d.state.image.Inspect()
	if info.Prefix.Length+info.Body.Length > MaxMutationBytes || d.TokenCount() > MaxMutationTokens {
		return 0, failure(MutationLimit, -1, info.Body)
	}
	if len(info.Blocks) != 2 || len(info.Packets) != 5 {
		return 0, failure(UnsupportedPrefix, -1, info.Prefix)
	}
	raw := d.state.image.Export()
	// The approved reader checks native identity, arithmetic, bounds, cycles and
	// delimiters. Consume its actual packet/block spans, never reparse a codec.
	first, last := info.Blocks[0], info.Blocks[1]
	if first.Data.Offset != 16 || first.Data.Length != 50 || last.Data.Length != 50 || first.Count != 5 || last.Count != 5 || first.Next != uint32(last.Data.Offset) || last.Next != 0 {
		return 0, failure(UnsupportedPrefix, -1, info.Prefix)
	}
	// Reader deliberately ignores unused slots/terminator payload; mutation may not.
	if !bytes.Equal(raw[last.Data.Offset+20:last.Data.Offset+50], make([]byte, 30)) {
		return 0, failure(UnsupportedPrefix, -1, last.Data)
	}
	types := [...]uint16{7, 15, 12, 6, 8}
	// WP42's complete pre-print field/extent proof qualifies one additional
	// linked index order. Physical payload offsets still come from Inspect.
	reordered := info.Packets[0].Type == 6
	if reordered {
		types = [...]uint16{6, 7, 15, 12, 8}
	}
	sizes := map[uint16]int{7: 14, 15: 86, 12: 90, 6: 16, 8: 2}
	spans := []wp51.Span{{Offset: 0, Length: 16}, first.Data, last.Data}
	flags := -1
	var pool, font []byte
	for i, p := range info.Packets {
		expectedIndex := first.Data.Offset + 10*(i+1)
		if i == 4 {
			expectedIndex = last.Data.Offset + 10
		}
		if p.Type != types[i] || p.Data.Length != sizes[p.Type] || p.Index != (wp51.Span{Offset: expectedIndex, Length: 10}) {
			return 0, failure(UnsupportedPrefix, -1, p.Index)
		}
		spans = append(spans, p.Data)
		packet := raw[p.Data.Offset : p.Data.Offset+p.Data.Length]
		category := UnsupportedResource
		if p.Type == 6 {
			category = UnsupportedMetadata
			flags = p.Data.Offset
		}
		if p.Type == 8 {
			category = UnsupportedPrefix
		}
		for _, f := range qualifiedPacketFields[p.Type] {
			actual := packet[f.offset : f.offset+len(f.value)]
			allowed := string(actual) == f.value
			if p.Type == 15 && f.offset == 21 && !reordered {
				allowed = allowed || string(actual) == "\x11\x72"
			}
			if p.Type == 6 && f.offset == 0 {
				allowed = actual[0] == 0 || actual[0] == 8
			}
			if !allowed {
				return 0, failure(category, -1, wp51.Span{Offset: p.Data.Offset + f.offset, Length: len(f.value)})
			}
		}
		if p.Type == 7 {
			pool = packet
		}
		if p.Type == 15 {
			font = packet
		}
		if p.Type == 8 && p.Data.Offset+p.Data.Length != info.Body.Offset {
			return 0, failure(UnsupportedPrefix, -1, p.Data)
		}
	}
	// All prefix bytes are owned exactly once, including physically relocated
	// packets. This derives prefix length; it never assumes B=324 or flags=256.
	sort.Slice(spans, func(i, j int) bool { return spans[i].Offset < spans[j].Offset })
	end := 0
	for _, span := range spans {
		if span.Offset != end {
			return 0, failure(UnsupportedPrefix, -1, span)
		}
		end += span.Length
	}
	if end != info.Body.Offset {
		return 0, failure(UnsupportedPrefix, -1, info.Prefix)
	}
	pointer := int(binary.LittleEndian.Uint16(font[18:20]))
	if pointer >= len(pool) || bytes.IndexByte(pool[pointer:], 0) < 0 {
		return 0, failure(UnsupportedResource, -1, info.Prefix)
	}
	// One font row. AFC FF=no chain, FE=unavailable; other values must index it.
	for _, ref := range font[50:86] {
		if ref != 254 && ref != 255 && ref >= 1 {
			return 0, failure(UnsupportedResource, -1, info.Prefix)
		}
	}
	if font[62] != 255 {
		return 0, failure(UnsupportedResource, -1, info.Prefix)
	}
	bold := false
	pairs := 0
	for i := 0; i < d.TokenCount(); i++ {
		span := d.span(i)
		switch identify(raw[span.Offset : span.Offset+span.Length]) {
		case ASCII:
		case BoldOn:
			if bold || pairs >= MaxBoldPairs {
				return 0, failure(UnsupportedBody, i, span)
			}
			bold = true
			pairs++
		case BoldOff:
			if !bold {
				return 0, failure(UnsupportedBody, i, span)
			}
			bold = false
		default:
			return 0, failure(UnsupportedBody, i, span)
		}
	}
	if bold {
		return 0, failure(UnsupportedBody, d.TokenCount(), info.Body)
	}
	return flags, nil
}

package document

import "github.com/lynncoleart/wordling/internal/wp51"

// Transient implementation policies, not original disk/memory capacity limits.
const (
	HistoryCapacity         = 3
	MaxHistoryFragmentBytes = MaxMutationBytes
)

type HistoryBoundary uint8

const (
	SavedBoundary HistoryBoundary = iota + 1
	RightBoundary
)

// History wraps private immutable state. NewHistory explicitly creates a fresh
// transient context; Parse, retrieve and clearing a document NEVER reset history.
// Cross-document continuation is unqualified and refuses rather than clearing it.
type History struct{ state *historyState }
type HistoryRevision struct{ state *historyState }
type historyState struct {
	document                *revisionState
	entries                 [HistoryCapacity]historyEntry
	count                   int
	groupOpen, groupUnknown bool
	effectUnqualified       bool
	groupGap                int
	seedView                bool
	seedStage               seedQualification
	observed                *historyObservation
}
type historyEntry struct {
	fragment fragment
	kind     DeletionKind
	commands int
}
type fragment struct {
	native      string
	kinds       []Kind
	codeIndices []uint16
}

func (f fragment) offset(index int) int {
	offset := index
	for _, code := range f.codeIndices {
		if int(code) < index {
			offset += 2
		}
	}
	return offset
}
func (d *Document) fragment(start, end int) fragment {
	span := wp51.Span{Offset: int(d.state.gaps[start]), Length: int(d.state.gaps[end] - d.state.gaps[start])}
	raw, _ := d.state.image.Bytes(span)
	f := fragment{native: string(raw), kinds: make([]Kind, end-start)}
	for i := start; i < end; i++ {
		s := d.span(i)
		local := s.Offset - span.Offset
		k := identify(raw[local : local+s.Length])
		f.kinds[i-start] = k
		if k == BoldOn || k == BoldOff {
			f.codeIndices = append(f.codeIndices, uint16(i-start))
		}
	}
	return f
}
func joinFragments(a, b fragment) fragment {
	f := fragment{native: a.native + b.native, kinds: make([]Kind, len(a.kinds)+len(b.kinds))}
	copy(f.kinds, a.kinds)
	copy(f.kinds[len(a.kinds):], b.kinds)
	f.codeIndices = append([]uint16(nil), a.codeIndices...)
	for _, index := range b.codeIndices {
		f.codeIndices = append(f.codeIndices, uint16(int(index)+len(a.kinds)))
	}
	return f
}
func NewHistory(d *Document) (History, error) {
	if !d.valid() {
		return History{}, failure(InvalidDocument, -1, noSpan)
	}
	if e := d.CheckMutation(); e != nil {
		return History{}, e
	}
	return History{&historyState{document: d.state}}, nil
}
func (h History) Revision() HistoryRevision { return HistoryRevision{h.state} }

// Qualified distinguishes enabled history behavior from preserved earlier
// entries after an edit whose immediate history effect has not been observed.
func (h History) Qualified() bool { return h.state != nil && !h.state.effectUnqualified }

// CountKnown reports whether stored fragment count also models actual stack
// count. Bounded observations deliberately withhold this stronger knowledge.
func (h History) CountKnown() bool {
	return h.state != nil && h.state.observed == nil && !h.state.effectUnqualified
}

// Count is stored fragment knowledge; CountKnown distinguishes an actual stack
// model from bounded WP40 observations with deliberately unknown hidden count.
func (h History) Count() int {
	if h.state == nil {
		return 0
	}

func (h History) check(d *Document) error {
	if h.state == nil {
		return failure(InvalidHistory, -1, noSpan)
	}
	if h.state.document != d.state {
		return failure(HistoryMismatch, -1, noSpan)
	}
	if h.state.effectUnqualified {
		return failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	return nil
}

// AfterEdit advances only a real immutable ASCII receipt. Native editing remains
// admitted independently of unobserved history effects: prior fragments survive,
// but subsequent history/deletion commands explicitly refuse after typeover
// replacement or insertion with existing history. The known Restore/reopen/X
// sequence remains admitted; no post-X history observation is invented.
func (h History) AfterEdit(r EditResult) (History, error) {
	if h.state == nil {
		return h, failure(InvalidHistory, -1, noSpan)
	}
	a := r.receipt
	if a == nil || !r.Document.valid() || r.Document.state != a.after || r.Cursor != a.cursor || r.Changed != a.changed {
		return h, failure(InvalidEditResult, -1, noSpan)
	}
	if a.before != h.state.document {
		return h, failure(HistoryMismatch, -1, noSpan)
	}
	if a.inputLength == 0 {
		return h, nil
	}
	next := *h.state
	next.document = a.after
	next.seedStage = seedUnqualified
	next.groupOpen = false
	next.groupUnknown = false
	if a.replaced > 0 || next.count > 0 {
		next.effectUnqualified = true
	}
	return History{&next}, nil
}

// EntryInfo is a copied inspection value; Native is original WP bytes, not labels.
type EntryInfo struct {
	Native           []byte
	Kind             DeletionKind
	Commands, Tokens int
	Qualified        bool
}

func (h History) Entry(index int) (EntryInfo, error) {
	if h.state == nil {
		return EntryInfo{}, failure(InvalidHistory, -1, noSpan)
	}
	if index < 0 || index >= h.state.count {
		return EntryInfo{}, failure(InvalidPosition, index, noSpan)
	}
	entry := h.state.entries[index]
	return EntryInfo{[]byte(entry.fragment.native), entry.kind, entry.commands, len(entry.fragment.kinds), h.Qualified()}, nil
}

// Separate records only the observed successful Save and Right grouping barriers.
// The dispatcher must report actual intervening commands; it may not reinterpret
// arbitrary keys/timing as these barriers or silently clear an existing stack.
func (h History) Separate(boundary HistoryBoundary) (History, error) {
	if h.state == nil {
		return h, failure(InvalidHistory, -1, noSpan)
	}
	if h.state.effectUnqualified || h.state.observed != nil {
		return h, failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	if boundary != SavedBoundary && boundary != RightBoundary {
		return h, failure(UnsupportedHistoryBoundary, -1, noSpan)
	}
	next := *h.state
	next.groupOpen = false
	next.groupUnknown = false
	if boundary == RightBoundary && (next.seedStage == seedBrowsed || next.seedStage == seedRightSeparated) {
		next.seedStage = seedRightSeparated
	} else {
		next.seedStage = seedUnqualified
	}
	return History{&next}, nil
}

// InterruptGrouping records a non-deleting interleaving whose grouping effect
// is unqualified (for example a view command between deletions). Entries and
// document state remain known; continuation refuses until an observed Save/Right
// separator. This is not a generic key router or a guessed new-deletion rule.
func (h History) InterruptGrouping() (History, error) {
	if h.state == nil {
		return h, failure(InvalidHistory, -1, noSpan)
	}
	if h.state.effectUnqualified || h.state.observed != nil {
		return h, failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	next := *h.state
	next.groupUnknown = next.groupUnknown || next.groupOpen
	next.groupOpen = false
	next.seedStage = seedUnqualified
	return History{&next}, nil
}
func (h History) record(s deletionState, f fragment, after *revisionState) (History, error) {
	if len(f.native) > MaxHistoryFragmentBytes || len(f.kinds) > MaxMutationTokens || len(f.codeIndices) > 2*MaxBoldPairs {
		return h, failure(MutationLimit, -1, noSpan)
	}
	next := *h.state
	next.document = after
	next.seedStage = seedUnqualified
	if s.kind == CharacterDeletion && s.target == 0 && h.state.count == 0 && len(f.kinds) == 1 && f.kinds[0] == ASCII && wp40Shape(after) {
		next.seedStage = seedAwaitingBrowse
		next.seedView = s.cursor.reveal
	}
	next.groupUnknown = false
	consecutive := s.kind == CharacterDeletion && s.direction == ForwardDelete && next.groupOpen && s.cursor.gap == next.groupGap && next.count > 0 && next.entries[0].kind == CharacterDeletion
	if consecutive {
		old := next.entries[0]
		if len(old.fragment.native) > MaxHistoryFragmentBytes-len(f.native) || len(old.fragment.kinds) > MaxMutationTokens-len(f.kinds) {
			return h, failure(MutationLimit, -1, noSpan)
		}
		old.fragment = joinFragments(old.fragment, f)
		old.commands++
		next.entries[0] = old
	} else {
		for i := HistoryCapacity - 1; i > 0; i-- {
			next.entries[i] = next.entries[i-1]
		}
		next.entries[0] = historyEntry{f, s.kind, 1}
		if next.count < HistoryCapacity {
			next.count++
		}
	}
	next.groupOpen = s.kind == CharacterDeletion && s.direction == ForwardDelete
	next.groupGap = s.target
	return History{&next}, nil
}

func (h History) Open(c Cursor) (Preview, error) {
	if h.state == nil {
		return Preview{}, failure(InvalidHistory, -1, noSpan)
	}
	if h.state.effectUnqualified {
		return Preview{}, failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	if h.state.observed != nil && (c.gap != h.state.observed.gap || c.reveal != h.state.observed.reveal) {
		return Preview{}, failure(UnsupportedHistoryEffect, c.gap, noSpan)
	}
	if c.state != h.state.document {
		return Preview{}, failure(RevisionMismatch, c.gap, noSpan)
	}
	if h.state.count == 0 {
		return Preview{}, nil
	}
	return Preview{&previewState{history: h.state, cursor: c}}, nil
}
func (p Preview) Previous() (Preview, error) {
	if !p.Active() {
		return p, failure(HistoryEmpty, -1, noSpan)
	}
	next := *p.state
	if next.history.observed != nil {
		if next.steps >= 3 {
			return p, failure(UnsupportedHistoryEffect, -1, noSpan)
		}
		next.index++
		next.steps++
	} else {
		next.index = (next.index + 1) % next.history.count
		next.steps++
	}
	return Preview{&next}, nil
}
func (h History) CancelPreview(p Preview) (History, error) {
	if !p.Active() {
		return h, failure(HistoryEmpty, -1, noSpan)
	}
	if h.state != p.state.history {
		return h, failure(HistoryMismatch, -1, noSpan)
	}
	next := *h.state
	if next.observed != nil {
		if p.state.steps != 3 {
			return h, failure(UnsupportedHistoryEffect, -1, noSpan)
		}
		observation := *next.observed
		observation.browsed = true
		next.observed = &observation
	}
	if next.seedStage == seedAwaitingBrowse && p.state.steps == 3 && p.state.cursor.gap == 0 && p.state.cursor.reveal == next.seedView {
		next.seedStage = seedBrowsed
	} else if next.seedStage != seedRightSeparated {
		next.seedStage = seedUnqualified
	}
	// No post-preview continued-delete grouping probe exists. Preserve entries;
	// refuse that unqualified interleaving until a known Save/Right separator.
	next.groupUnknown = next.groupUnknown || next.groupOpen
	next.groupOpen = false
	return History{&next}, nil
}
func (d *Document) Restore(h History, p Preview) (Deletion, error) {
	unchanged := unchangedDeletion(d, p.Cursor(), h)
	if !p.Active() {
		return unchanged, failure(HistoryEmpty, -1, noSpan)
	}
	if e := d.checkDeletion(p.Cursor(), h); e != nil {
		return unchanged, e
	}
	if p.state.history != h.state {
		return unchanged, failure(HistoryMismatch, -1, noSpan)
	}
	if h.state.observed != nil && (h.state.observed.restored || !h.state.observed.browsed || p.state.index != 0) {
		return unchanged, failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	f := h.state.entries[p.fragmentIndex()].fragment
	if len(f.kinds) > MaxMutationTokens-d.TokenCount() {
		return unchanged, failure(MutationLimit, p.Cursor().gap, noSpan)
	}
	raw := d.state.image.Export()
	start := int(d.state.gaps[0])
	at := int(d.state.gaps[p.Cursor().gap])
	bodyLength := len(raw) - start
	if len(f.native) > MaxMutationBytes-start-bodyLength {
		return unchanged, failure(MutationLimit, p.Cursor().gap, noSpan)
	}
	body := make([]byte, 0, bodyLength+len(f.native))
	body = append(body, raw[start:at]...)
	body = append(body, f.native...)
	body = append(body, raw[at:]...)
	edit, e := d.publishBody(p.Cursor(), body, d.TokenCount()+len(f.kinds), p.Cursor().gap+len(f.kinds), false)
	if e != nil {
		return unchanged, e
	}
	next := *h.state
	next.document = edit.Document.state
	if next.observed != nil {
		observation := *next.observed
		observation.restored = true
		observation.gap = edit.Cursor.gap
		observation.browsed = false
		next.observed = &observation
	}
	next.groupOpen = false
	next.groupUnknown = false
	// EC204 proves one attributed Restore, not its later stack lifetime. Keep
	// prior bytes but refuse subsequent queries; plain/opening-repair retention
	// is independently observed and remains enabled.
	if h.state.entries[p.fragmentIndex()].kind == AttributedDeletion {
		next.effectUnqualified = true
	}
	return Deletion{&deletionState{before: d.state, after: edit.Document.state, beforeHistory: h.state, afterHistory: &next, cursor: edit.Cursor, kind: Restoration, applied: true, bodyChanged: edit.Changed, nativeChanged: d.state != edit.Document.state}}, nil
}

	DeletionFragment

// Fragment spans are absent (Offset=-1): they are not current native-file bytes.
// FragmentStart/End identify virtual selection; the UI's [Block] marker is not a
// native token and must never be serialized into either document or history.
type OverlayElement struct {
	Origin     OverlayOrigin
	TokenIndex int
	Kind       Kind
	Span       wp51.Span
	Raw        []byte
	Text       string
}
type Overlay struct {
	Elements                                           []OverlayElement
	FragmentStart, FragmentEnd, CursorGap, TotalTokens int
	Reveal                                             bool
}

func (p Preview) Present(start, count int, reveal bool) (Overlay, error) {
	if !p.Active() {
		return Overlay{}, failure(HistoryEmpty, -1, noSpan)
	}
	d := &Document{state: p.state.history.document}
	f := p.state.history.entries[p.fragmentIndex()].fragment
	gap := p.state.cursor.gap
	total := d.TokenCount() + len(f.kinds)
	if start < 0 || start > total || count < 0 || count > total-start {
		return Overlay{}, failure(InvalidPosition, start, noSpan)
	}
	if count > MaxWindowTokens {
		return Overlay{}, failure(WindowLimit, start, noSpan)
	}
	out := Overlay{Elements: make([]OverlayElement, count), FragmentStart: gap, FragmentEnd: gap + len(f.kinds), CursorGap: gap, TotalTokens: total, Reveal: reveal}
	for i := start; i < start+count; i++ {
		var e OverlayElement
		if i >= gap && i < gap+len(f.kinds) {
			index := i - gap
			e = OverlayElement{Origin: DeletionFragment, TokenIndex: index, Kind: f.kinds[index], Span: noSpan, Raw: []byte(f.native[f.offset(index):f.offset(index+1)])}
		} else {
			index := i
			if i >= gap+len(f.kinds) {
				index -= len(f.kinds)
			}
			token, _ := d.Token(index)
			e = OverlayElement{Origin: DocumentToken, TokenIndex: index, Kind: token.Kind, Span: token.Span, Raw: token.Raw}
		}
		if e.Kind == ASCII {
			e.Text = string(e.Raw)
		} else if reveal {
			if e.Kind == BoldOn {
				e.Text = "[BOLD]"
			} else {
				e.Text = "[bold]"
			}
		}
		out.Elements[i-start] = e
	}
	return out, nil
}

// The qualification class fixes command order, view and token shape, while all
// character identities/fragments come from the actual immutable native values.
// Character values are not control commands or prerecorded-screen selectors.
func wp40Shape(state *revisionState) bool {
	d := &Document{state: state}
	if !d.valid() || d.TokenCount() != 6 {
		return false
	}
	for i, k := range []Kind{ASCII, BoldOn, ASCII, ASCII, BoldOff, ASCII} {
		token, _ := d.Token(i)
		if token.Kind != k {
			return false
		}
	}
	return true
}
func (h History) wp40Target(s deletionState) bool {
	if h.state == nil || h.state.observed != nil || h.state.seedStage != seedRightSeparated || h.state.seedView != s.cursor.reveal || !wp40Shape(h.state.document) {
		return false
	}
	if h.state.count != 1 || len(h.state.entries[0].fragment.kinds) != 1 || h.state.entries[0].fragment.kinds[0] != ASCII || h.state.entries[0].commands != 1 {
		return false
	}
	return s.kind == PairDeletion || (s.kind == ClosingRepair && s.blockStart == 4 && s.blockEnd == 5 && s.cursor.reveal)
}
func (h History) observeTarget(s deletionState, closing fragment, after *revisionState) History {
	next := *h.state
	next.document = after
	next.groupOpen = false
	next.groupUnknown = false
	next.seedStage = 0
	gap := s.target
	if s.kind == ClosingRepair {
		gap = s.blockEnd
	} else if s.target == s.off {
		gap--
	}
	next.observed = &historyObservation{closing: s.kind == ClosingRepair, gap: gap, reveal: s.cursor.reveal}
	if s.kind == ClosingRepair {
		next.entries[1] = next.entries[0]
		next.entries[0] = historyEntry{fragment: closing, kind: ClosingRepair, commands: 1}
		next.count = 2 // number of distinct stored observed fragments, NOT stack size.
	}
	return History{&next}
}

	if p.state.history.observed != nil {
		if p.state.history.observed.closing {
			return p.state.index % 2
		}
		return 0
	}
	return p.state.index

func (s Session) interrupted() (document.History, error) {
	if !s.historyValid || !s.history.Qualified() {

	}

}

// saveHistoryInterleaving separates known filename/publication behavior from
// optional hidden-history knowledge. Only Save receives this bounded waiver;
// other interleavings continue to use the dependency's exact guards.
func (s Session) saveHistoryInterleaving() (Session, error) {
	if !s.historyValid {
		return s, nil
	}
	h, err := s.history.InterruptGrouping()
	if err == nil {
		s.history = h
		return s, nil
	}
	var failure *document.Failure
	if !s.history.CountKnown() && errors.As(err, &failure) && failure.Category == document.UnsupportedHistoryEffect {
		// Bounded paired-code observations admit immediate history only; an
		// already-unqualified real history stays unavailable as well.
		// F10 is independently known, but its post-interleaving history is not.
		s.historyValid = false
		return s, nil
	}
	return s, err
}

type SaveState string

// Receipt preserves typed failures/residue, including detached late operations.
// Applied=false never means a filesystem action was rolled back or cancelled.
type Receipt struct {
	Result              Result
	Applied             bool
	DestinationMayExist bool
	Candidate           string
}

type Refusal struct{ Reason string }

func (r *Refusal) Error() string { return "editor: unsupported or invalid transition: " + r.Reason }
func refuse(reason string) error { return &Refusal{reason} }

func New(config Config) (Session, error) {
	locations := append([]Location(nil), config.Locations...)
	seen := map[string]bool{}
	for i, l := range locations {
		if l.Directory == nil || l.Directory.ID() == (storage.FileID{}) || seen[l.Prefix] {
			return Session{}, refuse("location requires unique prefix and pinned directory")
		}
		seen[l.Prefix] = true
		l.Directory = copyDirectory(l.Directory)
		locations[i] = l
	}
	native, err := document.New(document.NewConfig{Profile: document.HistoricalUS51DosTextStandardPrinter})
	if err != nil {
		return Session{}, err
	}
	cursor, err := native.CursorAt(0)
	if err != nil {
		return Session{}, err
	}
	history, err := document.NewHistory(native)
	if err != nil {
		return Session{}, err
	}
	return Session{initialized: true, transition: &transitionIdentity{1}, bufferID: &bufferIdentity{1}, screen: 1, macroRepeat: 1, locations: locations, mode: Editing, lastSave: NotSaved, native: copyNative(native), cursor: cursor, history: history, historyValid: true, genesisHistory: history.Revision(), newPhase: NewUnpublished, genesis: native.Revision(), revision: 1}, nil
}
func (s Session) Snapshot() Snapshot {
	x := Snapshot{Mode: s.mode, Name: s.name, Draft: s.draft, OriginalName: s.original, Notice: s.notice, Caret: s.caret, HomePrefix: s.homes, Gap: s.cursor.Gap(), Reveal: s.cursor.Reveal(), Revision: s.revision, SaveState: s.lastSave, Last: cloneFirstReceipt(s.last), Detached: cloneFirstReceipt(s.detached)}
	x.NewPhase, x.GenesisRevision = s.newPhase, s.genesis
	if s.first != nil {
		x.FirstPublication = firstReceipt(s.first.result, true, s.first.possible, s.first.candidate)
	}
	x.Dirty = s.dirty()
	x.Typeover, x.Selection = s.over, s.selection
	x.SelectionStart, x.SelectionEnd = s.anchor.Gap(), s.cursor.Gap()
	x.HistoryQualified, x.HistoryCountKnown, x.HistoryCount = s.historyValid && s.history.Qualified(), s.historyValid && s.history.CountKnown(), s.history.Count()
	x.PreviewIndex = s.preview.Index()
	x.NativeRevision, x.SavedRevision, x.PublishedRevision = s.native.Revision(), s.saved, s.publishedRevision
	x.PublishedName, x.Provenance = s.publishedName, s.provenance
	x.Published = receipt(s.published.Result, s.published.Applied)
	x.Pending = copyIntent(s.pending)
	x.Screen = s.screen
	x.MacroRepeat = s.macroRepeat
	if s.macroOwner != nil {
		x.MacroRun = s.macroOwner.run
	}
	switch s.mode {
	case DeleteQuestion:
		x.Prompt = s.deletion.Prompt()
	case HistoryPreview:
		x.Prompt = s.preview.Prompt()
	case MacroName:
		x.Prompt = "Macro: " + s.draft
	case RetrieveName:
		x.Prompt = "Document to be retrieved: " + s.draft + " (List Files)"
	case SaveName:
		x.Prompt = "Document to be saved: " + s.draft
	case ReplaceQuestion:
		x.Prompt = "Replace " + s.draft + "? No (Yes)"
	case ExitSaveQuestion:
		x.Prompt = "Save document? Yes (No)"
		if !s.dirty() {
			x.Notice = "(Text was not modified)"
		}
	case ExitQuestion:
		x.Prompt = "Exit WP? No (Yes) (Cancel to return to document)"
	}
	return x
}
func (s Session) Document() *document.Document { return copyNative(s.native) }
func (s Session) Source() *storage.Document    { return copySource(s.source) }
func (s Session) Present(start, count int) (document.Presentation, error) {
	if s.native == nil {
		return document.Presentation{}, refuse("no retrieved document")
	}
	return s.cursor.Present(start, count)
}
func (s Session) begin(kind IntentKind, dir *storage.Directory, name string) (Session, *Intent, error) {
	if kind == ReadMacro && s.native != nil && s.native.Birth().IsNew() && !s.newMacro.unedited(s) {
		return s, nil, refuse("New macro admission no longer owns naming target")
	}
	s.sequence++
	request := Intent{Operation: s.sequence, Revision: s.revision, Kind: kind, Directory: dir, Name: name, Document: s.source, Destination: s.target, Decision: storage.NoOverwrite}
	request.NativeRevision = s.native.Revision()
	if s.native != nil && kind != ReadMacro {
		request.Candidate, _ = s.native.Export()
	}
	if kind == ReadMacro {
		request.Document = nil
		request.Destination = nil
		request.MacroRun = request.Operation
		request.Repeat = s.macroRepeat
		request.Screen = s.screen
		request.macroProof = newMacroReadProof(request, s.bufferID)
		if s.newMacro != nil {
			request.macroProof.target = s.newMacro
			request.macroProof.issued = &transitionIdentity{1}
			request.macroProof.native = request.NativeRevision
		}
		s.macroRepeat = 1
	}
	s.pending = copyIntent(&request)
	s.resume = s.mode
	s.mode = Waiting
	s.notice = ""
	return s, copyIntent(s.pending), nil
}
func (s Session) resolve(name string) (*storage.Directory, string, error) {
	var best *Location
	for i := range s.locations {
		l := &s.locations[i]
		if strings.HasPrefix(name, l.Prefix) && (best == nil || len(l.Prefix) > len(best.Prefix)) {
			best = l
		}
	}
	if best == nil {
		return nil, "", refuse("filename is outside declared rooted locations")
	}
	leaf := strings.TrimPrefix(name, best.Prefix)
	if leaf == "" || leaf == "." || leaf == ".." || len(leaf) > storage.MaxNameBytes || strings.ContainsAny(leaf, "/\\:\x00") || strings.HasPrefix(leaf, ".wordling-candidate-") {
		return nil, "", refuse("only bounded flat basenames in a declared location")
	}
	return best.Directory, leaf, nil
}
func (s Session) filename(mode Mode) Session {
	s.mode = mode
	s.draft = s.name
	s.original = s.name
	s.caret = 0
	s.touched = false
	s.editedPrefill = false
	s.target = nil
	s.targetDir = nil
	s.targetName = ""
	s.notice = ""
	if mode == RetrieveName || mode == MacroName {
		s.draft = ""
		s.original = ""
	}
	return s
}
func (s Session) clear() Session {
	s.newPhase = ""
	s.first = nil
	s.genesis = document.Revision{}
	s.source = nil
	s.native = nil
	s.cursor = document.Cursor{}
	s.name = ""
	s.draft = ""
	s.original = ""
	s.caret = 0
	s.homes = 0
	s.revision++
	s.mode = Editing
	s.lastSave = NotSaved
	s.historyValid = false
	s.saved = document.Revision{}
	s.selection = false
	s.deletion = document.Deletion{}
	s.preview = document.Preview{}
	s.fromExit = false
	s.target = nil
	s.notice = ""
	return s
}

func (s Session) Reduce(event Event) (Session, *Intent, error) {
	if s.mode == Editing && s.macroRepeat != 1 && event.Command != Macro {
		return s, nil, refuse("qualified Macro repeat setup must be followed by Macro")
	}
	if s.macroOwner != nil && !s.macroDelivery {
		return s, nil, refuse("active macro owns delivery; use execution Step/Resume/HostStop")
	}

	next, intent, err := s.reduce(event)
	if err != nil && event.Command != Complete {
		return s, nil, err
	}
	if err == nil && event.Command != Complete {
		next.provenance = event.Command
		if intent != nil && intent.firstProof != nil {
			next.transition = intent.firstProof.transition
		} else if intent != nil && intent.macroProof != nil && intent.macroProof.target != nil {
			next.transition = intent.macroProof.issued
		} else {
			next = next.advanced()
		}
	}
	return next, intent, err
}
func (s Session) reduce(event Event) (Session, *Intent, error) {
	if !s.initialized {
		return s, nil, refuse("uninitialized session")
	}
	if event.Command == Complete {
		if event.Result == nil || event.Text != "" {
			return s, nil, refuse("completion requires result only")
		}
		return s.complete(*event.Result)
	}
	if event.Result != nil || (event.Command != Text && event.Text != "") {
		return s, nil, refuse("mixed event payload")
	}
	if s.newPhase == PublishedUnbound {
		if event.Command == RetryFirstRead {
			return s.retryFirstRead()
		}
		return s, nil, refuse("published New requires Read-only binding retry")
	}
	if s.newPhase == PublicationUncertain {
		return s, nil, refuse("New publication effect unresolved; host reconciliation required")
	}
	if s.mode == Exited {
		return s, nil, refuse("session exited")
	}
	if s.mode == Waiting {
		if s.pending != nil && (s.pending.Kind == PublishCandidate || s.pending.Kind == WriteSnapshot) && editingCommand(event.Command) {
			old := s
			s.mode = Editing
			next, _, err := s.reduce(event)
			if err != nil {
				return old, nil, err
			}
			next.mode = Waiting
			return next, nil, nil
		}
		if event.Command != CancelOperation {
			return s, nil, refuse("operation pending; host cancellation is separate from original keys")
		}
		if s.pending != nil && s.pending.firstProof != nil {
			p := s.pending.firstProof
			if p.kind == PublishNewBuffer {
				s.newPhase = PublicationUncertain
				s.first = &firstPublication{request: p, result: Result{Kind: p.kind, Operation: p.operation, Revision: p.revision}, possible: true}
				s.lastSave = SaveUncertain
			} else {
				s.newPhase = PublishedUnbound
				s.lastSave = SaveUncertain
			}
		}
		s.pending = nil
		s.newMacro = nil
		s.mode = s.resume
		s.notice = "Operation detached; filesystem completion still requires reconciliation"
		return s, nil, nil
	}
	if s.mode == MacroError {
		if event.Command == F1 {
			s.mode = Editing
			s.notice = ""
			return s, nil, nil
		}
		return s, nil, refuse("missing macro recovery requires observed F1")
	}
	if s.mode == MacroReady && event.Command == CancelOperation {
		s.newMacro = nil
		s.mode = Editing
		s.notice = "Host discarded unstarted macro; source snapshot retained"
		return s, nil, nil
	}
	if event.Command == CancelOperation {
		return s, nil, refuse("no pending operation")
	}
	if s.mode == DeleteQuestion || s.mode == HistoryPreview {
		return s.edit(event)
	}
	if s.mode == RetrieveName || s.mode == SaveName || s.mode == MacroName {
		return s.editName(event)
	}
	switch s.mode {
	case ReplaceQuestion:
		switch event.Command {
		case No:
			s.mode = SaveName
			s.caret = 0
			s.touched = false
			s.target = nil
			return s, nil, nil
		case Yes:
			if s.source == nil {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "New replacement", Name: s.targetName}
			}

			raw, _ := s.native.Export()
			if !bytes.Equal(raw, s.source.Export()) {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "edited replacement", Name: s.targetName}
			}
			identity, _ := s.target.Identity()
			if identity.ID != s.source.Identity().ID {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "session replacement decision", Name: s.targetName}
			}
			next, _, err := s.begin(WriteSnapshot, s.targetDir, s.targetName)
			request := *next.pending
			request.Decision = storage.AllowOverwrite
			next.pending = &request
			return next, copyIntent(next.pending), err
		}
	case ExitSaveQuestion:
		switch event.Command {
		case F1:
			s.mode = Editing
			return s, nil, nil
		case Escape:
			s = s.filename(SaveName)
			s.fromExit = true
			return s, nil, nil
		case No:
			s.mode = ExitQuestion
			return s, nil, nil
		}
	case ExitQuestion:
		switch event.Command {
		case F1:
			s.mode = Editing
			return s, nil, nil
		case No, Escape:
			return s.clear(), nil, nil
		case Yes:
			s = s.clear()
			s.mode = Exited
			return s, nil, nil
		}
	case Editing:
		if s.selection && event.Command != Right && event.Command != Backspace && event.Command != F1 {
			return s, nil, refuse("only forward block movement/Backspace and qualified F1 cleanup")
		}
		if s.homes > 0 && event.Command != Home && event.Command != Up {
			return s, nil, refuse("unobserved Home-prefix interruption")
		}
		switch event.Command {
		case SwitchDocument:
			return s.switchDocument()
		case Macro:
			if s.native == nil || s.selection || s.homes != 0 {
				return s, nil, refuse("macro invocation requires qualified loaded editing context")
			}
			newTarget := s.native.Birth().IsNew()
			if newTarget && !s.pristineNewMacro() {
				return s, nil, refuse("macro requires pristine initial New Doc1")
			}
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			if newTarget {
				s.newMacro = s.captureNewMacro()
			}
			return s.filename(MacroName), nil, nil
		case Retrieve:
			if s.source != nil || (s.native != nil && s.native.Revision() != s.genesis) {
				return s, nil, refuse("insertion into an existing buffer is unimplemented")
			}
			return s.filename(RetrieveName), nil, nil
		case Save:
			if s.source == nil && s.newPhase != NewUnpublished {
				return s, nil, refuse("new document writer unavailable")
			}
			next, err := s.saveHistoryInterleaving()
			if err != nil {
				return s, nil, err
			}
			s = next
			s.fromExit = false
			return s.filename(SaveName), nil, nil
		case F7:
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			s.mode = ExitSaveQuestion
			return s, nil, nil
		case Text, Insert, Delete, Backspace, Block, F1, Previous, Restore:
			return s.edit(event)
		case Reveal:
			if s.native == nil {
				return s, nil, refuse("Reveal requires retrieved document")
			}
			if s.selection {
				return s, nil, refuse("view change during block is unqualified")
			}
			if s.historyValid && s.history.Qualified() {
				h, err := s.history.InterruptGrouping()
				if err != nil {
					return s, nil, err
				}
				s.history = h
			}
			s.cursor = s.cursor.WithReveal(!s.cursor.Reveal())
			return s, nil, nil
		case Left, Right:
			if s.native == nil {
				return s, nil, refuse("cursor requires retrieved document")
			}
			direction := 1
			if event.Command == Left {
				direction = -1
			}
			cursor, err := s.cursor.Move(direction)
			if err != nil {
				return s, nil, err
			}
			if s.historyValid && s.history.Qualified() {
				var h document.History
				if event.Command == Right {
					h, err = s.history.Separate(document.RightBoundary)
				} else {
					h, err = s.history.InterruptGrouping()
				}
				if err != nil {
					return s, nil, err
				}
				s.history = h
			}
			s.cursor = cursor
			s.provenance = event.Command
			return s, nil, nil
		case Home:
			if s.native == nil || s.homes == 3 {
				return s, nil, refuse("Home prefix outside observed bounded sequence")
			}
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			s.homes++
			return s, nil, nil
		case Up:
			if s.homes != 2 && s.homes != 3 {
				return s, nil, refuse("only Home2/3 Up endpoints observed")
			}
			gap := 0
			if s.native.TokenCount() > 0 {
				token, _ := s.native.Token(0)
				if token.Kind == document.Opaque {
					return s, nil, refuse("opaque initial code semantics unknown")
				}
				if s.homes == 2 && token.Kind == document.BoldOn {
					gap = 1
				}
			}
			cursor, err := s.native.CursorAt(gap)
			if err != nil {
				return s, nil, err
			}
			s.cursor = cursor.WithReveal(s.cursor.Reveal())
			s.homes = 0
			return s, nil, nil
		}
	}
	return s, nil, refuse(fmt.Sprintf("command %d in %s", event.Command, s.mode))
}

func (s Session) editName(event Event) (Session, *Intent, error) {
	switch event.Command {
	case Macro:
		if s.mode == SaveName {
			return s, nil, nil
		}
		return s, nil, refuse("macro nesting in this filename prompt unqualified")
	case F1:
		s.newMacro = nil
		s.mode = Editing
		s.fromExit = false
		s.notice = ""
		return s, nil, nil
	case Escape:
		if s.mode == RetrieveName {
			s.mode = Editing
			s.notice = ""
			return s, nil, nil
		}
	case Text:
		if len(event.Text) > storage.MaxNameBytes {
			return s, nil, refuse("filename draft safety bound")
		}
		if event.Text == "" {
			return s, nil, refuse("empty filename text")
		}
		for _, r := range event.Text {
			if r < 0x20 || r > 0x7e {
				return s, nil, refuse("filename requires explicit ASCII profile")
			}
		}
		draft, caret := s.draft, s.caret
		if !s.touched {
			s.editedPrefill = false
			draft = ""
			caret = 0
		}
		if len(draft)+len(event.Text) > storage.MaxNameBytes {
			return s, nil, refuse("filename draft safety bound")
		}
		s.draft = draft[:caret] + event.Text + draft[caret:]
		s.caret = caret + len(event.Text)
		s.touched = true
		return s, nil, nil
	case Right:
		if s.mode == RetrieveName || s.mode == MacroName {
			return s, nil, refuse("Retrieve filename editing unobserved")
		}
		if s.caret >= len(s.draft) {
			return s, nil, refuse("filename caret boundary")
		}
		if !s.touched && s.draft != "" {
			s.editedPrefill = true
		}
		s.caret++
		s.touched = true
		return s, nil, nil
	case Backspace:
		if s.mode == RetrieveName || s.mode == MacroName {
			return s, nil, refuse("Retrieve filename editing unobserved")
		}
		if s.caret == 0 {
			return s, nil, refuse("filename caret boundary")
		}
		s.draft = s.draft[:s.caret-1] + s.draft[s.caret:]
		s.caret--
		s.touched = true
		return s, nil, nil
	case Enter:
		if s.mode == SaveName && s.editedPrefill {
			return s, nil, refuse("prefilled filename edit has only cancel evidence")
		}
		if s.fromExit {
			return s, nil, refuse("save-on-exit completion not observed; F1 can return to document")
		}
		resolve := s.resolve
		if s.mode == MacroName {
			resolve = s.resolveMacro
		}
		dir, name, err := resolve(s.draft)
		if err != nil {
			return s, nil, err
		}
		if s.mode == MacroName {
			return s.begin(ReadMacro, dir, name)
		}
		if s.mode == RetrieveName {
			return s.begin(Read, dir, name)
		}
		if s.source == nil && s.newPhase != NewUnpublished {
			return s, nil, refuse("new document writer unavailable")
		}
		s.lastSave = NotSaved
		return s.begin(Observe, dir, name)
	}
	return s, nil, refuse("unobserved filename command")
}

func receipt(result Result, applied bool) Receipt {
	result = copyResult(result)
	record := Receipt{Result: result, Applied: applied, DestinationMayExist: (!applied || result.Err != nil) && result.Outcome.State == storage.Published}
	var failure *storage.Failure
	if errors.As(result.Err, &failure) {
		copy := *failure
		record.Result.Err = &copy
		// A typed storage failure is authoritative about its transaction stage.
		record.DestinationMayExist = failure.DestinationMayExist
		record.Candidate = failure.Candidate
	}
	return record
}
func (s Session) complete(result Result) (Session, *Intent, error) {
	result = copyResult(result)
	if s.pending != nil && s.pending.firstProof != nil {
		if result.Operation != s.pending.Operation || result.Revision != s.pending.Revision || result.Kind != s.pending.Kind {
			s.detached = receipt(result, false)
			return s, nil, nil
		}
		return s.completeFirst(result)
	}

				return invalid("invalid same-file no-op result")
			}
			s.lastSave = SameFileNoOp
		case storage.Published:
			// A real published candidate has an inode. Device zero can be valid;
			// this structural check cannot authenticate arbitrary host IO claims.
			if out.Identity.ID.Inode == 0 {
				return invalid("published destination has no inode identity")
			}
			if request.Destination.Exists() {
				return invalid("existing destination cannot be published")
			}
			s.lastSave = NewNamePublished
		default:
			return invalid("unknown save outcome")
		}
		s.mode = Editing
		s.name = s.draft
		s.saved = request.NativeRevision
		s.publishedRevision, s.publishedName = request.NativeRevision, s.name
		s.published = receipt(result, true)
		if s.historyValid {
			h, err := s.history.Separate(document.SavedBoundary)
			if err != nil {
				// Publication already happened. Preserve its receipt and native
				// revision, but never retain permission after a failed barrier.
				s.historyValid = false
				s.notice = err.Error()
				return s, nil, err
			}
			s.history = h
		}
		s.notice = ""
	default:
		return invalid("unknown operation result")
	}
	return s, nil, nil
}

func freezeFirstError(err error, invoked bool) (*sealedError, bool, string) {
	if err == nil {
		return nil, false, ""
	}
	out := &sealedError{}
	possible, candidate := false, ""
	budget := 64
	var walk func(error, bool)
	walk = func(e error, covered bool) {
		if e == nil {
			return
		}
		budget--
		if budget < 0 {
			possible = true
			return
		}
		if f, ok := e.(*storage.Failure); ok && f != nil {
			v := *f
			v.Cause = nil
			out.failures = append(out.failures, v)

			if candidate == "" {
				candidate = v.Candidate
			}
			switch v.Kind {
			case storage.InvalidRequest, storage.UnsafePath, storage.ReadFailed, storage.InputLimit, storage.NativeRefused, storage.StaleSource, storage.DestinationChanged, storage.OverwriteRequired, storage.UnsupportedReplacement, storage.CandidateFailed, storage.PublicationFailed, storage.CleanupFailed:
			default:
				possible = true
			}
			possible = possible || v.DestinationMayExist
			walk(f.Cause, true)
			return
		}
		if many, ok := e.(interface{ Unwrap() []error }); ok {
			for _, child := range many.Unwrap() {
				walk(child, covered)
			}
			return
		}
		if one, ok := e.(interface{ Unwrap() error }); ok {
			walk(one.Unwrap(), covered)
			return
		}
		if !covered {
			possible = true
		}
		out.message += e.Error() + "; "
	}
	walk(err, false)
	if !invoked {
		possible = false
	}
	return out, possible, candidate
}

		return m, nil
	}

		result := reply.result
		if firstKind(result.Kind) {
			m.notice = "Unbound first-save payload refused; authentic receipt retained"
			return m, nil
		}
		if result.Kind == editor.ReadMacro {
			m.notice = "Unbound macro IO payload refused; authentic receipt retained"
			return m, nil
		}
		remaining := make([]operation, 0, len(m.unsettled))

		for _, op := range m.unsettled {
			if op == (operation{result.Operation, result.Revision, result.Kind}) {

			} else {
				remaining = append(remaining, op)
			}
		}
		m.unsettled = remaining
		if warning := Reconciliation(result); warning != "" {
			m.reconcile = warning
		}
		pending := m.session.Snapshot().Pending
		attached := pending != nil && pending.Operation == result.Operation && pending.Revision == result.Revision && pending.Kind == result.Kind
		if !attached && result.Err == nil && result.Outcome.State == storage.Published {
			m.reconcile = "Detached save published a destination; filesystem reconciliation required"
		}
		return m.reduce(editor.Event{Command: editor.Complete, Result: &result})
	}
	if _, ok := msg.(CancelPendingMsg); ok {
		return m.reduce(editor.Event{Command: editor.CancelOperation})
	}
	event := input.Translate(msg)
	if event.Kind == input.Resize {
		m.menu = menuDisplay{}
		m.width = max(1, min(event.Width, 4096))
		m.height = max(1, min(event.Height, 4096))
		return m, nil
	}
	if event.Kind != input.Press {
		switch event.Kind {
		case input.Paste, input.PasteStart, input.PasteEnd, input.Unknown:
			m.notice = "Unsupported event: " + string(event.Kind) + " (" + event.SourceType + ")"
		}
		return m, nil
	}
	m.invalidateMenu()
	if next, handled := m.menuPress(*event.Key); handled {
		return next, nil
	}
	if m.reconcile != "" {
		m.notice = "Resolve filesystem uncertainty before further commands"
		return m, nil
	}
	if (len(m.unsettled) > 0 || m.firstPending != nil) && m.session.Snapshot().Mode != editor.Waiting {
		m.notice = "Detached IO still pending; awaiting its actual receipt"
		return m, nil
	}
	k := event.Key
	if k.IsRepeat {
		m.notice = "Unsupported repeated key event"
		return m, nil
	}
	// Contextual Linux verification retry, not an original WP command claim.
	if k.Code == tea.KeyF10 && k.Mod == 0 && m.session.Snapshot().NewPhase == editor.PublishedUnbound {
		return m.reduce(editor.Event{Command: editor.RetryFirstRead})
	}
	if next, cmd, handled := m.macroKey(k.Code, k.Mod, k.Text); handled {
		return next, cmd
	}
	command, ok := resolve(*k, m.session.Snapshot().Mode)
	if !ok {
		m.notice = "Unsupported key (physical delivery unqualified)"
		return m, nil
	}
	return m.reduce(command)
}
func resolve(k input.Key, mode editor.Mode) (editor.Event, bool) {
	// Delivered text keeps its identity. No q/CtrlQ/CtrlS command aliases.
	if k.Mod&^(tea.ModShift|tea.ModCapsLock) == 0 && k.Text != "" {
		switch mode {
		case editor.RetrieveName, editor.SaveName, editor.MacroName, editor.Editing, editor.Waiting:
			return editor.Event{Command: editor.Text, Text: k.Text}, true
		case editor.ReplaceQuestion, editor.ExitSaveQuestion, editor.ExitQuestion, editor.DeleteQuestion:
			if k.Text == "y" || k.Text == "Y" {
				return editor.Event{Command: editor.Yes}, true
			}
			if k.Text == "n" || k.Text == "N" {
				return editor.Event{Command: editor.No}, true
			}
		case editor.HistoryPreview:
			if k.Text == "1" {
				return editor.Event{Command: editor.Restore}, true
			}
			if k.Text == "2" {
				return editor.Event{Command: editor.Previous}, true
			}
		}
	}
	event := editor.Event{}
	switch {
	case k.Code == tea.KeyF10 && k.Mod == tea.ModAlt:
		event.Command = editor.Macro
	case k.Code == tea.KeyF3 && k.Mod == tea.ModShift:
		event.Command = editor.SwitchDocument
	case k.Code == tea.KeyF10 && k.Mod == tea.ModShift:
		event.Command = editor.Retrieve
	case k.Code == tea.KeyF3 && k.Mod == tea.ModAlt:
		event.Command = editor.Reveal
	case k.Code == tea.KeyF4 && k.Mod == tea.ModAlt:
		event.Command = editor.Block
	case k.Mod != 0:
		return event, false
	default:
		switch k.Code {
		case tea.KeyF10:
			event.Command = editor.Save
		case tea.KeyF1:
			event.Command = editor.F1
		case tea.KeyF7:
			event.Command = editor.F7
		case tea.KeyEscape:
			event.Command = editor.Escape
		case tea.KeyHome:
			event.Command = editor.Home
		case tea.KeyUp:
			event.Command = editor.Up
		case tea.KeyRight:
			event.Command = editor.Right
		case tea.KeyLeft:
			event.Command = editor.Left
		case tea.KeyBackspace:
			event.Command = editor.Backspace
		case tea.KeyDelete:
			event.Command = editor.Delete
		case tea.KeyInsert:
			event.Command = editor.Insert
		case tea.KeyEnter:
			event.Command = editor.Enter

		}

			n.WriteString("\x1b[1m")
			rt = "[BOLD]"
		case document.BoldOff:
			n.WriteString("\x1b[22m")
			rt = "[bold]"
		case document.ASCII:

			}

		}

		}

		// returned by the authoritative native Present can own active cells.
		if nativeActive && e.TokenIndex == p.CursorGap && e.Span.Length > 0 {
			role = revealActive
		}

		if i < gap {
			rx += len(rt)
		}

		lower = append(lower, revealRun{text: " ", role: revealActive})
	}

	ns, rs := max(0, nx-max(1, m.width/2)), max(0, rx-max(1, m.width/2))

func clip(s string, width int) string {

	return out

		return nil, failure(InvalidNewEncoding, -1, noSpan)
	}
	printer.u16(date)
	printer.u16(clock) // cached resource stamp, never host/FAT time
	printer.u16(600)
	printer.u8(0x40) // point-size/typeface tail in native12 (different SET54h order)
	state := newEncoder{}
	state.u8(0)
	state.u8(0x23)
	state.u16(0x7c) // document flags, text/graphics quality, redline
	state.u16(uint16(init.screenWidth))
	state.u16(0) // scoped screen width/binding WPU
	state.u8(init.printerCounter)
	state.u8(0) // scoped counter/reserved
	state.u16(init.displayPitchRaw)
	state.u16(init.resolutionRaw)
	state.u16(0) // raw encodings/reserved
	graphics := newEncoder{}
	graphics.u16(0) // no graphics payload
	packets := []struct {
		kind uint16
		data *newEncoder
	}{{7, &pool}, {15, &font}, {12, &printer}, {6, &state}, {8, &graphics}}
	const headerBytes = 16
	const indexBytes = 10
	const entriesPerBlock = 5
	blockBytes := entriesPerBlock * indexBytes
	lastBlock := headerBytes + blockBytes
	for _, p := range packets[:4] {
		if p.data.err {
			return nil, failure(InvalidNewEncoding, -1, noSpan)
		}
		lastBlock += len(p.data.bytes)
	}
	bodyStart := lastBlock + blockBytes + len(graphics.bytes)
	if graphics.err || bodyStart > MaxNewPrefixBytes || bodyStart > MaxMutationBytes {
		return nil, failure(InvalidNewEncoding, -1, noSpan)
	}
	w := newEncoder{bytes: make([]byte, 0, bodyStart)}
	w.u8(0xff)
	w.u8('W')
	w.u8('P')
	w.u8('C')
	w.u32(uint32(bodyStart))
	w.u8(1)
	w.u8(0x0a)
	w.u8(0)
	w.u8(1)
	w.u16(0)
	w.u16(0) // identity/version/encryption/reserved
	w.u16(0xfffb)
	w.u16(entriesPerBlock)
	w.u16(uint16(blockBytes))
	w.u32(uint32(lastBlock))
	offset := headerBytes + blockBytes
	for _, p := range packets[:4] {
		w.u16(p.kind)
		w.u32(uint32(len(p.data.bytes)))
		w.u32(uint32(offset))
		offset += len(p.data.bytes)
	}
	for _, p := range packets[:4] {
		w.add(p.data.bytes...)
	}
	w.u16(0xfffb)
	w.u16(entriesPerBlock)
	w.u16(uint16(blockBytes))
	w.u32(0)
	w.u16(8)
	w.u32(uint32(len(graphics.bytes)))
	w.u32(uint32(lastBlock + blockBytes))
	for i := 2 * indexBytes; i < blockBytes; i++ {
		w.u8(0)
	} // wholly zero unused slots
	w.add(graphics.bytes...)
	if w.err || len(w.bytes) != bodyStart || binary.LittleEndian.Uint32(w.bytes[4:8]) != uint32(len(w.bytes)) {
		return nil, failure(InvalidNewEncoding, -1, noSpan)
	}
	return w.bytes, nil
}
