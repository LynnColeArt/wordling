sha=lambda b:hashlib.sha256(b).hexdigest()
checks=[];logs=[];refs=[]
def check(name,ok,detail=None):
 checks.append({'check':name,'pass':bool(ok),'detail':detail})
 if not ok: raise AssertionError(name)
def git(*args):
 r=subprocess.run(['git',*args],cwd=R,capture_output=True);logs.append({'argv':['git',*args],'exit':r.returncode,'stderr':r.stderr.decode()});check('git exit',r.returncode==0,args);return r.stdout
def source(ref,path):
 b=git('show',ref+':'+path);refs.append({'commit':ref,'path':path,'bytes':len(b),'sha256':sha(b),'blob':git('rev-parse',ref+':'+path).decode().strip()});return b
