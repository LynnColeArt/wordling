func (m Model) Session() editor.Session { return m.session }
func (m Model) Init() tea.Cmd           { return nil }
func (m Model) reduce(event editor.Event) (tea.Model, tea.Cmd) {
	previousMode := m.session.Snapshot().Mode
	next, intent, err := m.session.Reduce(event)
	m.session = next
	if err == nil && next.Snapshot().Mode == editor.MacroName && previousMode != editor.MacroName {
		m.macroStatus = MacroStatus{}
		m.macroFrozen = nil
	}
	m.notice = ""
	if err != nil {
		m.notice = err.Error()
	}
	if intent != nil {
		if firstKind(intent.Kind) {
			return m.queueFirst(*intent)
		}
		m.unsettled = append(append([]operation(nil), m.unsettled...), operation{intent.Operation, intent.Revision, intent.Kind})
		return m, m.runtime.command(*intent)
	}
	if next.Snapshot().Mode == editor.Exited {
		return m, tea.Quit
	}
	return m, nil
}
func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if m.runtime == nil {
		m.notice = "Uninitialized host runtime"
		return m, nil
	}
	token, valid := m.runtime.claimUI(m.uiAuthority)
	if !valid {
		m.notice = "Stale UI transition refused; authentic model retains authority"
		return m, nil
	}
	m.uiAuthority = token
	next, cmd := m.update(msg)
	model := next.(Model)
	model.invalidateMenu()
	model.prepareASCIIViewport()
	return model, cmd
}
func (m Model) update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg.(type) {
	case firstReply, RetryFirstReadMsg, macroIOReply, macroReply, macroDuplicateMsg, macroReadRefusedMsg, HostStopMacroMsg, duplicateDispatchMsg, resultMsg, CancelPendingMsg:
		m.menu = menuDisplay{}
	}
	switch v := msg.(type) {
	case tea.EnvMsg, tea.ColorProfileMsg:
		// Bubble Tea startup administration is not delivered user input.
		return m, nil
	case firstReply:
		return m.firstResult(v.ticket)
	case RetryFirstReadMsg:
		if m.firstPending != nil || len(m.unsettled) > 0 || m.macro != nil {
			m.notice = "Actual IO still pending; Read retry refused"
			return m, nil
		}
		return m.reduce(editor.Event{Command: editor.RetryFirstRead})
	case macroIOReply:
		return m.macroIO(v.receipt)
	case macroReply:
		return m.macroReply(v)
	case macroDuplicateMsg:
		m.notice = "Duplicate macro dispatch refused; authentic continuation retained"
		return m, nil
	case macroReadRefusedMsg:
		return m.macroReadRefused(v)
	case HostStopMacroMsg:
		return m.hostStopMacro(v.Reason)
	}
	if _, ok := msg.(duplicateDispatchMsg); ok {
		m.notice = "Duplicate IO dispatch refused; no additional operation executed"
		return m, nil
	}
	if reply, ok := msg.(resultMsg); ok {
		result := reply.result
		if firstKind(result.Kind) {
			m.notice = "Unbound first-save payload refused; authentic receipt retained"
			return m, nil
		}
		if result.Kind == editor.ReadMacro {
			m.notice = "Unbound macro IO payload refused; authentic receipt retained"
			return m, nil
		}
		remaining := make([]operation, 0, len(m.unsettled))

		for _, op := range m.unsettled {
			if op == (operation{result.Operation, result.Revision, result.Kind}) {

			} else {
				remaining = append(remaining, op)
			}
		}
		m.unsettled = remaining
		if warning := Reconciliation(result); warning != "" {
			m.reconcile = warning
		}
		pending := m.session.Snapshot().Pending
		attached := pending != nil && pending.Operation == result.Operation && pending.Revision == result.Revision && pending.Kind == result.Kind
		if !attached && result.Err == nil && result.Outcome.State == storage.Published {
			m.reconcile = "Detached save published a destination; filesystem reconciliation required"
		}
		return m.reduce(editor.Event{Command: editor.Complete, Result: &result})
	}
	if _, ok := msg.(CancelPendingMsg); ok {
		return m.reduce(editor.Event{Command: editor.CancelOperation})
	}
	event := input.Translate(msg)
	if event.Kind == input.Resize {
		m.menu = menuDisplay{}
		m.width = max(1, min(event.Width, 4096))
		m.height = max(1, min(event.Height, 4096))
		return m, nil
	}
	if event.Kind != input.Press {
		switch event.Kind {
		case input.Paste, input.PasteStart, input.PasteEnd, input.Unknown:
			m.notice = "Unsupported event: " + string(event.Kind) + " (" + event.SourceType + ")"
		}
		return m, nil
	}
	m.invalidateMenu()
	if next, handled := m.menuPress(*event.Key); handled {
		return next, nil
	}
	if m.reconcile != "" {
		m.notice = "Resolve filesystem uncertainty before further commands"
		return m, nil
	}
	if (len(m.unsettled) > 0 || m.firstPending != nil) && m.session.Snapshot().Mode != editor.Waiting {
		m.notice = "Detached IO still pending; awaiting its actual receipt"
		return m, nil
	}
	k := event.Key
	if k.IsRepeat {
		m.notice = "Unsupported repeated key event"
		return m, nil
	}
	// Contextual Linux verification retry, not an original WP command claim.
	if k.Code == tea.KeyF10 && k.Mod == 0 && m.session.Snapshot().NewPhase == editor.PublishedUnbound {
		return m.reduce(editor.Event{Command: editor.RetryFirstRead})
	}
	if next, cmd, handled := m.macroKey(k.Code, k.Mod, k.Text); handled {
		return next, cmd
	}
	command, ok := resolve(*k, m.session.Snapshot().Mode)
	if !ok {
		m.notice = "Unsupported key (physical delivery unqualified)"
		return m, nil
	}
	return m.reduce(command)
}
func resolve(k input.Key, mode editor.Mode) (editor.Event, bool) {
	// Delivered text keeps its identity. No q/CtrlQ/CtrlS command aliases.
	if k.Mod&^(tea.ModShift|tea.ModCapsLock) == 0 && k.Text != "" {
		switch mode {
		case editor.RetrieveName, editor.SaveName, editor.MacroName, editor.Editing, editor.Waiting:
			return editor.Event{Command: editor.Text, Text: k.Text}, true
		case editor.ReplaceQuestion, editor.ExitSaveQuestion, editor.ExitQuestion, editor.DeleteQuestion:
			if k.Text == "y" || k.Text == "Y" {
				return editor.Event{Command: editor.Yes}, true
			}
			if k.Text == "n" || k.Text == "N" {
				return editor.Event{Command: editor.No}, true
			}
		case editor.HistoryPreview:
			if k.Text == "1" {
				return editor.Event{Command: editor.Restore}, true
			}
			if k.Text == "2" {
				return editor.Event{Command: editor.Previous}, true
			}
		}
	}
	event := editor.Event{}
	switch {
	case k.Code == tea.KeyF10 && k.Mod == tea.ModAlt:
		event.Command = editor.Macro
	case k.Code == tea.KeyF3 && k.Mod == tea.ModShift:
		event.Command = editor.SwitchDocument
	case k.Code == tea.KeyF10 && k.Mod == tea.ModShift:
		event.Command = editor.Retrieve
	case k.Code == tea.KeyF3 && k.Mod == tea.ModAlt:
		event.Command = editor.Reveal
	case k.Code == tea.KeyF4 && k.Mod == tea.ModAlt:
		event.Command = editor.Block
	case k.Mod != 0:
		return event, false
	default:
		switch k.Code {
		case tea.KeyF10:
			event.Command = editor.Save
		case tea.KeyF1:
			event.Command = editor.F1
		case tea.KeyF7:
			event.Command = editor.F7
		case tea.KeyEscape:
			event.Command = editor.Escape
		case tea.KeyHome:
			event.Command = editor.Home
		case tea.KeyUp:
			event.Command = editor.Up
		case tea.KeyRight:
			event.Command = editor.Right
		case tea.KeyLeft:
			event.Command = editor.Left
		case tea.KeyBackspace:
			event.Command = editor.Backspace
		case tea.KeyDelete:
			event.Command = editor.Delete
		case tea.KeyInsert:
			event.Command = editor.Insert
		case tea.KeyEnter:
			event.Command = editor.Enter
		default:
			return event, false
		}
	}
	return event, true
}

func (m Model) legacyPanes() (normal, reveal string, x int, err error) {
	d := m.session.Document()
	if d == nil {
		return "", "", 0, nil
	}
	s := m.session.Snapshot()
	gap := s.Gap
	if gap > document.MaxWindowTokens {
		return "", "", 0, fmt.Errorf("cursor beyond bounded %d-token presentation", document.MaxWindowTokens)
	}
	count := min(d.TokenCount(), document.MaxWindowTokens)
	p, err := m.session.Present(0, count)
	if err != nil {
		return "", "", 0, err
	}
	elements := p.Elements
	blockStart, blockEnd := -1, -1
	if s.Selection {
		blockStart, blockEnd = s.SelectionStart, s.SelectionEnd
	}
	if s.Mode == editor.HistoryPreview {
		info, e := m.session.Preview(0, 0)
		if e != nil {
			return "", "", 0, e
		}
		if info.TotalTokens > document.MaxWindowTokens {
			return "", "", 0, fmt.Errorf("preview exceeds bounded %d-token presentation", document.MaxWindowTokens)
		}
		overlay, e := m.session.Preview(0, info.TotalTokens)
		if e != nil {
			return "", "", 0, e
		}
		elements = make([]document.Element, len(overlay.Elements))
		for i, v := range overlay.Elements {
			elements[i] = document.Element{TokenIndex: i, Kind: v.Kind, Text: v.Text}
		}
		blockStart, blockEnd = overlay.FragmentStart, overlay.FragmentEnd
	}
	var n strings.Builder
	var lower []revealRun
	nativeActive := !s.Selection && s.Mode != editor.HistoryPreview
	nx, rx := 0, 0
	for i := 0; i <= len(elements); i++ {
		if i == blockStart {
			n.WriteString("[Block]")
			lower = append(lower, revealRun{text: "[Block]", role: revealBase})
			if i < gap {
				nx += 7
				rx += 7
			}
		}
		if i == blockEnd {
			n.WriteString("[Block]")
			lower = append(lower, revealRun{text: "[Block]", role: revealBase})
			if i < gap {
				nx += 7
				rx += 7
			}
		}
		if i == len(elements) {
			break
		}
		e := elements[i]
		rt := e.Text
		switch e.Kind {
		case document.BoldOn:
			n.WriteString("\x1b[1m")
			rt = "[BOLD]"
		case document.BoldOff:
			n.WriteString("\x1b[22m")
			rt = "[bold]"
		case document.ASCII:
			n.WriteString(e.Text)
			if i < gap {
				nx += len(e.Text)
			}
		default:
			return "", "", 0, fmt.Errorf("unsupported token presentation")
		}
		role := revealBase
		if e.Kind == document.BoldOn || e.Kind == document.BoldOff {
			role = revealCode
		}
		// Overlay elements deliberately have no native span. Only an element
		// returned by the authoritative native Present can own active cells.
		if nativeActive && e.TokenIndex == p.CursorGap && e.Span.Length > 0 {
			role = revealActive
		}
		lower = append(lower, revealRun{text: rt, role: role})
		if i < gap {
			rx += len(rt)
		}
	}
	if nativeActive && p.CursorGap == d.TokenCount() {
		lower = append(lower, revealRun{text: " ", role: revealActive})
	}
	// Terminal cell scrolling is view-only, never original pagination or Pos.
	ns, rs := max(0, nx-max(1, m.width/2)), max(0, rx-max(1, m.width/2))
	x = nx - ns
	return ansi.TruncateLeft(n.String()+"\x1b[0m", ns, ""), renderReveal(lower, rs, m.width), x, nil
}
func clip(s string, width int) string {
	out := ansi.Truncate(s, width, "")
	if strings.Contains(s, "\x1b[") {
		out += "\x1b[0m"
	}
	return out
}
func (m Model) View() tea.View {
	s := m.session.Snapshot()
	frame := m.visibleFrame()
	if frame.anchors {
		return m.anchorView(frame, s.Reveal)
	}
	normal, reveal, x, err := frame.normal, frame.reveal, frame.x, frame.err
	lines := []string{"Wordling 5.2 | bounded native editing", normal}
	if err != nil {
		lines[1] = "Unsupported document presentation: " + err.Error()
	}
	if s.Reveal {
		banner := "Press Reveal Codes to restore screen"
		if frame.inspection.Partial {
			banner = "Partial Reveal inspection | HOST ?N = unsupported owner"
		}
		lines = append(lines, banner, reveal)
	}
	prompt := s.Prompt
	if m.repeatEntry {
		prompt = "Repeat Value = "
	} else if s.Mode == editor.Editing && s.MacroRepeat == 3 {
		prompt = "Repeat Value = 3"
	}
	prompt = m.menuPrompt(s, prompt)
	lines = append(lines, prompt, s.Notice)
	if line := m.macroLine(); line != "" {
		lines = append(lines, line)
	}
	if s.Screen == 2 {
		lines = append(lines, "Doc2")
	}
	if line := firstStateLine(s); line != "" {
		lines = append(lines, line)
	}
	if s.Mode == editor.Waiting {
		lines = append(lines, "Filesystem operation pending")
	}
	if m.notice != "" {
		lines = append(lines, m.notice)
	}
	if m.reconcile != "" {
		lines = append(lines, m.reconcile)
	}
	history := "unknown"
	if s.HistoryCountKnown {
		history = fmt.Sprint(s.HistoryCount)
	}
	typing := ""
	if s.Typeover {
		typing = "Typeover "
	}
	profileLabel := "local-flat: ASCII basenames only; physical key delivery UNQUALIFIED"
	if m.runtime != nil && m.runtime.profile == MacroCOUT {
		profileLabel = "macro-cout: exact C:\\OUT\\ and flat names; physical key delivery UNQUALIFIED"
	}
	lines = append(lines, fmt.Sprintf("%s | %s | gap %d | %s", s.Mode, s.Name, s.Gap, s.SaveState),
		fmt.Sprintf("%sDirty=%t Block=%t revision=%d history=%s qualified=%t", typing, s.Dirty, s.Selection, s.Revision, history, s.HistoryQualified), profileLabel)
	if m.menuState().Visible {
		lines[0] = renderMenuBar()
	}
	for i := range lines {
		lines[i] = clip(lines[i], m.width)
	}
	if len(lines) > m.height {
		lines = lines[:m.height]
	}
	v := tea.NewView(strings.Join(lines, "\n"))
	v.AltScreen = true
	v.KeyboardEnhancements = tea.KeyboardEnhancements{ReportEventTypes: true, ReportAlternateKeys: true, ReportAllKeysAsEscapeCodes: true, ReportAssociatedText: true}
	if (s.Mode == editor.Editing || s.Mode == editor.DeleteQuestion || s.Mode == editor.HistoryPreview || s.Mode == editor.Waiting) && err == nil && m.height > 1 {
		y := 1
		if y < m.height {
			v.Cursor = tea.NewCursor(min(x, m.width-1), y)
		}
	}
	// Prompt caret refers to the original reducer draft, not document position.
	if (s.Mode == editor.RetrieveName || s.Mode == editor.SaveName || s.Mode == editor.MacroName) && len(lines) > 2 {
		prefix := "Document to be retrieved: "
		if s.Mode == editor.SaveName {
			prefix = "Document to be saved: "
		} else if s.Mode == editor.MacroName {
			prefix = "Macro: "
		}
		y := 2
		if s.Reveal {
			y = 4
		}
		if y < m.height {
			v.Cursor = tea.NewCursor(min(len(prefix)+s.Caret, m.width-1), y)
		}
	}
	if (m.repeatEntry || s.Mode == editor.Editing && s.MacroRepeat == 3) && m.height > 2 {
		x := len("Repeat Value = ")
		if !m.repeatEntry {
			x++
		}
		v.Cursor = tea.NewCursor(min(x, m.width-1), 2)
	}
	return v
}

// anchorEligible observes existing source/UI owners. It grants no command or IO
// authority. The pristine menu uses its qualified legacy base on BOTH toggles.
func (m Model) anchorEligible(s editor.Snapshot, frame macroPane) bool {
	if m.width != 80 || m.height != 25 || m.runtime == nil || frame.err != nil || frame.inspection.Partial ||
		s.Mode != editor.Editing || s.Selection || s.Typeover || !s.HistoryCountKnown || s.Screen != 1 || s.MacroRepeat != 1 ||
		s.Pending != nil || s.Prompt != "" || s.Notice != "" || m.repeatEntry || m.notice != "" || m.reconcile != "" ||
		len(m.unsettled) != 0 || m.firstPending != nil || m.macro != nil || m.macroTicket != nil || m.macroCPU ||
		m.macroDeferred != nil || m.macroStatus.Active || m.macroStatus.WorkerPending || m.macroStatus.DisplaySuppressed ||
		s.NewPhase == editor.PublishedUnbound || s.NewPhase == editor.PublicationUncertain ||
		m.menuReason() == "" || m.menu.diagnostic != "" {
		return false
	}
	m.runtime.mu.Lock()
	defer m.runtime.mu.Unlock()
	return !m.runtime.stopping && m.runtime.macro == nil && m.uiAuthority != nil && m.uiAuthority == m.runtime.uiAuthority
}

// A fixed bounded display composition, not original page/ruler/status layout.
// Frozen frames retain this choice even when later live state/size is different.
func (m Model) anchorView(frame macroPane, reveal bool) tea.View {
	lines := make([]string, 25)
	lines[0] = revealBase + frame.normal
	if reveal {
		lines[13] = frame.reveal
		lines[24] = renderReveal([]revealRun{
			{text: "Press ", role: revealBase},
			{text: "Reveal Codes", role: revealCode},
			{text: " to restore screen" + strings.Repeat(" ", 44), role: revealBase},
		}, 0, 80)
	}
	for i := range lines {
		lines[i] = clip(lines[i], m.width)
	}
	lines = lines[:min(len(lines), m.height)]
	v := tea.NewView(strings.Join(lines, "\n"))
	v.AltScreen = true
	v.KeyboardEnhancements = tea.KeyboardEnhancements{ReportEventTypes: true, ReportAlternateKeys: true, ReportAllKeysAsEscapeCodes: true, ReportAssociatedText: true}
	v.Cursor = tea.NewCursor(min(frame.x, m.width-1), 0)
	return v
}

# Bounded executable native editing composition (WP18/WP28)

Run `go run ./cmd/wordling -root /absolute/local/directory -profile local-flat`
from a terminal. The operator chooses a validated local filesystem; the app does
not certify a network mount or interpret DOS drives. Only flat ASCII basenames
are supported. Shift+F10 retrieves, Alt+F3 toggles Reveal Codes, F10 invokes the
approved snapshot or native-candidate save workflow, and F7 invokes the original bounded exit prompts.
No global q/Ctrl+Q/Ctrl+S/Escape aliases exist. F1/Escape retain their observed
context-specific reducer meaning. Missing/unobserved branches refuse explicitly.

`Open(Config{Root, Profile: LocalFlat})` returns a directory-owning Runtime and
value Model. Model implements the pinned Bubble Tea v2.1.1 interface. Update
consumes input.Translate and calls the actual editor.Session.Reduce; View consumes
shared document.Presentation token positions. State() and Session() expose
immutable inspection for tests/measurement. Keep one authoritative model chain;
do not execute intents from forked old values. No terminal code serializes native
bytes or decides original page geometry. The normal pane styles supported Bold;
the Reveal pane shows the same tokens and gap. The presentation has a4096-token
safety window and horizontally clips terminal cells; it does not paginate.
Cursor beyond that window and opaque-token windows explicitly refuse presentation
while preserving the complete native snapshot for no-edit storage.

Production uses only the empty rooted prefix. Text Shift/CapsLock modifiers may
supply uppercase filenames; command modifiers remain exact. Paste, releases,
repeats, unknown messages and unsupported keys never replay editing commands.
Unknown/paste states are visible. Resize changes bounded terminal dimensions
(1..4096 cells per axis), never the document. Physical delivery remains UNQUALIFIED
in State and the application view; requesting keyboard enhancements is separate
from actual terminal capability or physical evidence. No original Pos inches,
line/page metrics, DOS raster, help/menu engine or unsupported writer is invented.
Original prompt wording and unmodified/Reveal status cues are preserved; their
row/spacing arrangement follows this bounded Linux presentation, not an accepted
pixel/whole-screen compatibility claim.

Each tea.Cmd captures one complete immutable editor.Intent. Runtime calls the
actual Directory.Read/Observe/Save/PreparePublication/Publish and returns the exact operation/revision/kind.
The reducer validates document/directory/name/native fingerprint and source
identity. Observe and Save are separate operations, never a claimed atomic pair.
Same-source explicit Yes can yield SameFileNoOp; a new name yields
NewNamePublished only after real verified IO. Existing different inodes always
refuse replacement, even for identical bytes. Source is not rebound after Save As.

CancelPendingMsg is an explicitly programmatic host detach, with no physical key
alias or claim of disk cancellation. Further keys are held until its actual late
receipt arrives; detached publication requires filesystem reconciliation and
never becomes current success. Typed cleanup/uncertain failures retain candidate
and destination-may-exist details, block further commands, preserve source/native
state, and never auto-exit. No speculative rollback/unlink/recovery action occurs.
The owner must inspect filesystem state after abnormal termination; there is no
fake acknowledgement button claiming recovery.

Runtime.Close gates new operations under the same lock as worker registration,
drains every started call, then closes the pinned directory. Queued commands
executed afterward refuse before touching descriptors. Each operation envelope
can execute only once. A duplicate dispatch returns a distinct non-completion
notice, checked before shutdown/capacity; it cannot settle the admitted operation
or consume its pending receipt. A first dispatch rejected before admission by
shutdown/capacity returns a definitive failure because no worker owns that
operation. A1024-operation host safety cap bounds receipt/identity
history (not an original limit). Completed() returns copied outcome/error metadata
without retaining loaded native images/destination handles. The actual command
checks terminal restoration, waits for worker drain, checks receipts/residue, and
reports host signals as abnormal rather than original successful exit. SIGKILL,
power-loss durability and automatic startup recovery remain unqualified. Runtime
Close does not promise a time bound on a blocked external filesystem.

Tests use real temporary directories and actual executable PTYs; injectable
completion errors explicitly test the host result boundary, not filesystem fault
claims. Approved storage tests cover lower-level faults. No native framing or

// from the approved E API; H neither reads native bytes nor reclassifies owners.
func inspectionRuns(c document.CatalogueInspection) ([]revealRun, error) {
	if len(c.Elements) > document.MaxWindowTokens {
		return nil, fmt.Errorf("inspection render owner limit")
	}
	size := 0
	for _, e := range c.Elements {
		n := len(e.Text)
		if e.Kind == document.Opaque && e.Text == "" {
			if e.TokenIndex < 0 {
				return nil, fmt.Errorf("inspection invalid owner ordinal")
			}
			n = 2
			for value := e.TokenIndex; value >= 10; value /= 10 {
				n++
			}
		}
		if n > MaxInspectionRenderBytes-size {
			return nil, fmt.Errorf("inspection expanded render limit")
		}
		size += n
	}
	runs := make([]revealRun, 0, len(c.Elements))
	for _, e := range c.Elements {
		text, role := e.Text, revealBase
		if e.Kind == document.Opaque && text == "" {
			text = "?" + strconv.Itoa(e.TokenIndex)
		} else if e.Kind != document.ASCII {
			role = revealCode
		}
		runs = append(runs, revealRun{text: text, role: role})
	}
	return runs, nil
}

package tui

import "strings"

// Logical Linux ANSI profile derived from the selected original VGA roles:
// 17 = normal white on blue; 1F = bright white on blue; 47 = normal white on red.
// Terminal palette/RGB/raster identity is not implied. These are view values,
// never token identities, native bytes or an independent document model.
const (
	revealBase   = "\x1b[22;37;44m"
	revealCode   = "\x1b[22;97;44m"
	revealActive = "\x1b[22;37;41m"
)

type revealRun struct{ text, role string }

// Present's admitted ASCII and the derived code/Block labels are single-cell
// ASCII. Intersect display runs BEFORE emitting escapes, so clipping cannot cut
// an SGR or lose the selected role when its first cells are left of the viewport.
func renderReveal(runs []revealRun, left, width int) string {
	var out strings.Builder
	position := 0
	lastRole := ""
	for _, run := range runs {
		end := position + len(run.text)
		startCell, endCell := max(left, position), min(left+width, end)
		if startCell < endCell {
			if run.role != lastRole {
				out.WriteString(run.role)
				lastRole = run.role
			}
			out.WriteString(run.text[startCell-position : endCell-position])
		}
		position = end
		if position >= left+width {
			break
		}
	}
	out.WriteString("\x1b[0m")
	return out.String()
}

		m.ascii = asciiQualification{}
		return
	}

	if d == nil || d.TokenCount() <= document.MaxWindowTokens {

		return
	}

		return
	}
	m.ascii = qualifyASCII(d, s.NativeRevision)
}
func qualifyASCII(d *document.Document, revision document.Revision) asciiQualification {
	total := d.TokenCount()
	fact := asciiQualification{revision: revision, total: total, checked: true}
	// Preflight total before requesting any catalogue allocation. This is only a
	// HOST view limit, never a native open/edit limit.
	if total <= document.MaxWindowTokens || total > maxASCIIViewportOwners {
		return fact
	}
	rawBytes, textBytes := 0, 0
	previousEnd := -1
	for start := 0; start < total; start += document.MaxWindowTokens {
		count := min(document.MaxWindowTokens, total-start)
		c, err := document.InspectRevealCatalogue(d, revision, start, count)
		if err != nil || c.Revision != revision || c.StartGap != start || c.EndGap != start+count || c.TotalTokens != total || c.TrueEOF != (start+count == total) || len(c.Elements) != count {
			return fact
		}
		for i, e := range c.Elements {
			if e.TokenIndex != start+i || e.Kind != document.ASCII || e.Span.Length != 1 || len(e.Text) != 1 || e.Text[0] < 32 || e.Text[0] > 126 || e.Span.Offset < 0 || (previousEnd >= 0 && e.Span.Offset != previousEnd) {
				return fact
			}
			rawBytes += e.Span.Length
			textBytes += len(e.Text)
			previousEnd = e.Span.Offset + e.Span.Length
			if rawBytes > 1<<20 || textBytes > 128<<10 {
				return fact
			}
		}
	}
	fact.whole = true
	return fact
}

}

	editedPrefill             bool
	pending                   *Intent
	target                    *storage.Destination
	targetDir                 *storage.Directory
	targetName                string
	lastSave                  SaveState
	published                 Receipt
	last, detached            Receipt
	notice                    string
	history                   document.History
	historyValid, historyUsed bool
	over                      bool

}

func New(config Config) (Session, error) {
	locations := append([]Location(nil), config.Locations...)
	seen := map[string]bool{}
	for i, l := range locations {
		if l.Directory == nil || l.Directory.ID() == (storage.FileID{}) || seen[l.Prefix] {
			return Session{}, refuse("location requires unique prefix and pinned directory")
		}
		seen[l.Prefix] = true
		l.Directory = copyDirectory(l.Directory)
		locations[i] = l
	}
	native, err := document.New(document.NewConfig{Profile: document.HistoricalUS51DosTextStandardPrinter})
	if err != nil {
		return Session{}, err
	}
	cursor, err := native.CursorAt(0)
	if err != nil {
		return Session{}, err
	}
	history, err := document.NewHistory(native)
	if err != nil {
		return Session{}, err
	}
	return Session{initialized: true, transition: &transitionIdentity{1}, bufferID: &bufferIdentity{1}, screen: 1, macroRepeat: 1, locations: locations, mode: Editing, lastSave: NotSaved, native: copyNative(native), cursor: cursor, history: history, historyValid: true, genesisHistory: history.Revision(), newPhase: NewUnpublished, genesis: native.Revision(), revision: 1}, nil
}
func (s Session) Snapshot() Snapshot {
	x := Snapshot{Mode: s.mode, Name: s.name, Draft: s.draft, OriginalName: s.original, Notice: s.notice, Caret: s.caret, HomePrefix: s.homes, Gap: s.cursor.Gap(), Reveal: s.cursor.Reveal(), Revision: s.revision, SaveState: s.lastSave, Last: cloneFirstReceipt(s.last), Detached: cloneFirstReceipt(s.detached)}
	x.NewPhase, x.GenesisRevision = s.newPhase, s.genesis
	if s.first != nil {
		x.FirstPublication = firstReceipt(s.first.result, true, s.first.possible, s.first.candidate)
	}
	x.Dirty = s.dirty()
	x.Typeover, x.Selection = s.over, s.selection
	x.SelectionStart, x.SelectionEnd = s.anchor.Gap(), s.cursor.Gap()
	x.HistoryQualified, x.HistoryCountKnown, x.HistoryCount = s.historyValid && s.history.Qualified(), s.historyValid && s.history.CountKnown(), s.history.Count()
	x.PreviewIndex = s.preview.Index()
	x.NativeRevision, x.SavedRevision, x.PublishedRevision = s.native.Revision(), s.saved, s.publishedRevision
	x.PublishedName, x.Provenance = s.publishedName, s.provenance
	x.Published = receipt(s.published.Result, s.published.Applied)
	x.Pending = copyIntent(s.pending)
	x.Screen = s.screen
	x.MacroRepeat = s.macroRepeat
	if s.macroOwner != nil {
		x.MacroRun = s.macroOwner.run
	}
	switch s.mode {
	case DeleteQuestion:
		x.Prompt = s.deletion.Prompt()
	case HistoryPreview:
		x.Prompt = s.preview.Prompt()
	case MacroName:
		x.Prompt = "Macro: " + s.draft
	case RetrieveName:
		x.Prompt = "Document to be retrieved: " + s.draft + " (List Files)"
	case SaveName:
		x.Prompt = "Document to be saved: " + s.draft
	case ReplaceQuestion:
		x.Prompt = "Replace " + s.draft + "? No (Yes)"
	case ExitSaveQuestion:
		x.Prompt = "Save document? Yes (No)"
		if !s.dirty() {
			x.Notice = "(Text was not modified)"
		}
	case ExitQuestion:
		x.Prompt = "Exit WP? No (Yes) (Cancel to return to document)"
	}
	return x
}
func (s Session) Document() *document.Document { return copyNative(s.native) }
func (s Session) Source() *storage.Document    { return copySource(s.source) }
func (s Session) Present(start, count int) (document.Presentation, error) {
	if s.native == nil {
		return document.Presentation{}, refuse("no retrieved document")
	}
	return s.cursor.Present(start, count)
}
func (s Session) begin(kind IntentKind, dir *storage.Directory, name string) (Session, *Intent, error) {
	if kind == ReadMacro && s.native != nil && s.native.Birth().IsNew() && !s.newMacro.unedited(s) {
		return s, nil, refuse("New macro admission no longer owns naming target")
	}
	s.sequence++
	request := Intent{Operation: s.sequence, Revision: s.revision, Kind: kind, Directory: dir, Name: name, Document: s.source, Destination: s.target, Decision: storage.NoOverwrite}
	request.NativeRevision = s.native.Revision()
	if s.native != nil && kind != ReadMacro {
		request.Candidate, _ = s.native.Export()
	}
	if kind == ReadMacro {
		request.Document = nil
		request.Destination = nil
		request.MacroRun = request.Operation
		request.Repeat = s.macroRepeat
		request.Screen = s.screen
		request.macroProof = newMacroReadProof(request, s.bufferID)
		if s.newMacro != nil {
			request.macroProof.target = s.newMacro
			request.macroProof.issued = &transitionIdentity{1}
			request.macroProof.native = request.NativeRevision
		}
		s.macroRepeat = 1
	}
	s.pending = copyIntent(&request)
	s.resume = s.mode
	s.mode = Waiting
	s.notice = ""
	return s, copyIntent(s.pending), nil
}
func (s Session) resolve(name string) (*storage.Directory, string, error) {
	var best *Location
	for i := range s.locations {
		l := &s.locations[i]
		if strings.HasPrefix(name, l.Prefix) && (best == nil || len(l.Prefix) > len(best.Prefix)) {
			best = l
		}
	}
	if best == nil {
		return nil, "", refuse("filename is outside declared rooted locations")
	}
	leaf := strings.TrimPrefix(name, best.Prefix)
	if leaf == "" || leaf == "." || leaf == ".." || len(leaf) > storage.MaxNameBytes || strings.ContainsAny(leaf, "/\\:\x00") || strings.HasPrefix(leaf, ".wordling-candidate-") {
		return nil, "", refuse("only bounded flat basenames in a declared location")
	}
	return best.Directory, leaf, nil
}
func (s Session) filename(mode Mode) Session {
	s.mode = mode
	s.draft = s.name
	s.original = s.name
	s.caret = 0
	s.touched = false
	s.editedPrefill = false
	s.target = nil
	s.targetDir = nil
	s.targetName = ""
	s.notice = ""
	if mode == RetrieveName || mode == MacroName {
		s.draft = ""
		s.original = ""
	}
	return s
}
func (s Session) clear() Session {
	s.newPhase = ""
	s.first = nil
	s.genesis = document.Revision{}
	s.source = nil
	s.native = nil
	s.cursor = document.Cursor{}
	s.name = ""
	s.draft = ""
	s.original = ""
	s.caret = 0
	s.homes = 0
	s.revision++
	s.mode = Editing
	s.lastSave = NotSaved
	s.historyValid = false
	s.saved = document.Revision{}
	s.selection = false
	s.deletion = document.Deletion{}
	s.preview = document.Preview{}
	s.fromExit = false
	s.target = nil
	s.notice = ""
	return s
}

func (s Session) Reduce(event Event) (Session, *Intent, error) {
	if s.mode == Editing && s.macroRepeat != 1 && event.Command != Macro {
		return s, nil, refuse("qualified Macro repeat setup must be followed by Macro")
	}
	if s.macroOwner != nil && !s.macroDelivery {
		return s, nil, refuse("active macro owns delivery; use execution Step/Resume/HostStop")
	}

	next, intent, err := s.reduce(event)
	if err != nil && event.Command != Complete {
		return s, nil, err
	}
	if err == nil && event.Command != Complete {
		next.provenance = event.Command
		if intent != nil && intent.firstProof != nil {
			next.transition = intent.firstProof.transition
		} else if intent != nil && intent.macroProof != nil && intent.macroProof.target != nil {
			next.transition = intent.macroProof.issued
		} else {
			next = next.advanced()
		}
	}
	return next, intent, err
}
func (s Session) reduce(event Event) (Session, *Intent, error) {
	if !s.initialized {
		return s, nil, refuse("uninitialized session")
	}
	if event.Command == Complete {
		if event.Result == nil || event.Text != "" {
			return s, nil, refuse("completion requires result only")
		}
		return s.complete(*event.Result)
	}
	if event.Result != nil || (event.Command != Text && event.Text != "") {
		return s, nil, refuse("mixed event payload")
	}
	if s.newPhase == PublishedUnbound {
		if event.Command == RetryFirstRead {
			return s.retryFirstRead()
		}
		return s, nil, refuse("published New requires Read-only binding retry")
	}
	if s.newPhase == PublicationUncertain {
		return s, nil, refuse("New publication effect unresolved; host reconciliation required")
	}
	if s.mode == Exited {
		return s, nil, refuse("session exited")
	}
	if s.mode == Waiting {
		if s.pending != nil && (s.pending.Kind == PublishCandidate || s.pending.Kind == WriteSnapshot) && editingCommand(event.Command) {
			old := s
			s.mode = Editing
			next, _, err := s.reduce(event)
			if err != nil {
				return old, nil, err
			}
			next.mode = Waiting
			return next, nil, nil
		}
		if event.Command != CancelOperation {
			return s, nil, refuse("operation pending; host cancellation is separate from original keys")
		}
		if s.pending != nil && s.pending.firstProof != nil {
			p := s.pending.firstProof
			if p.kind == PublishNewBuffer {
				s.newPhase = PublicationUncertain
				s.first = &firstPublication{request: p, result: Result{Kind: p.kind, Operation: p.operation, Revision: p.revision}, possible: true}
				s.lastSave = SaveUncertain
			} else {
				s.newPhase = PublishedUnbound
				s.lastSave = SaveUncertain
			}
		}
		s.pending = nil
		s.newMacro = nil
		s.mode = s.resume
		s.notice = "Operation detached; filesystem completion still requires reconciliation"
		return s, nil, nil
	}
	if s.mode == MacroError {
		if event.Command == F1 {
			s.mode = Editing
			s.notice = ""
			return s, nil, nil
		}
		return s, nil, refuse("missing macro recovery requires observed F1")
	}
	if s.mode == MacroReady && event.Command == CancelOperation {
		s.newMacro = nil
		s.mode = Editing
		s.notice = "Host discarded unstarted macro; source snapshot retained"
		return s, nil, nil
	}
	if event.Command == CancelOperation {
		return s, nil, refuse("no pending operation")
	}
	if s.mode == DeleteQuestion || s.mode == HistoryPreview {
		return s.edit(event)
	}
	if s.mode == RetrieveName || s.mode == SaveName || s.mode == MacroName {
		return s.editName(event)
	}
	switch s.mode {
	case ReplaceQuestion:
		switch event.Command {
		case No:
			s.mode = SaveName
			s.caret = 0
			s.touched = false
			s.target = nil
			return s, nil, nil
		case Yes:
			if s.source == nil {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "New replacement", Name: s.targetName}
			}

			raw, _ := s.native.Export()
			if !bytes.Equal(raw, s.source.Export()) {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "edited replacement", Name: s.targetName}
			}
			identity, _ := s.target.Identity()
			if identity.ID != s.source.Identity().ID {
				return s, nil, &storage.Failure{Kind: storage.UnsupportedReplacement, Stage: "session replacement decision", Name: s.targetName}
			}
			next, _, err := s.begin(WriteSnapshot, s.targetDir, s.targetName)
			request := *next.pending
			request.Decision = storage.AllowOverwrite
			next.pending = &request
			return next, copyIntent(next.pending), err
		}
	case ExitSaveQuestion:
		switch event.Command {
		case F1:
			s.mode = Editing
			return s, nil, nil
		case Escape:
			s = s.filename(SaveName)
			s.fromExit = true
			return s, nil, nil
		case No:
			s.mode = ExitQuestion
			return s, nil, nil
		}
	case ExitQuestion:
		switch event.Command {
		case F1:
			s.mode = Editing
			return s, nil, nil
		case No, Escape:
			return s.clear(), nil, nil
		case Yes:
			s = s.clear()
			s.mode = Exited
			return s, nil, nil
		}
	case Editing:
		if s.selection && event.Command != Right && event.Command != Backspace && event.Command != F1 {
			return s, nil, refuse("only forward block movement/Backspace and qualified F1 cleanup")
		}
		if s.homes > 0 && event.Command != Home && event.Command != Up {
			return s, nil, refuse("unobserved Home-prefix interruption")
		}
		switch event.Command {
		case SwitchDocument:
			return s.switchDocument()
		case Macro:
			if s.native == nil || s.selection || s.homes != 0 {
				return s, nil, refuse("macro invocation requires qualified loaded editing context")
			}
			newTarget := s.native.Birth().IsNew()
			if newTarget && !s.pristineNewMacro() {
				return s, nil, refuse("macro requires pristine initial New Doc1")
			}
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			if newTarget {
				s.newMacro = s.captureNewMacro()
			}
			return s.filename(MacroName), nil, nil
		case Retrieve:
			if s.source != nil || (s.native != nil && s.native.Revision() != s.genesis) {
				return s, nil, refuse("insertion into an existing buffer is unimplemented")
			}
			return s.filename(RetrieveName), nil, nil
		case Save:
			if s.source == nil && s.newPhase != NewUnpublished {
				return s, nil, refuse("new document writer unavailable")
			}
			next, err := s.saveHistoryInterleaving()
			if err != nil {
				return s, nil, err
			}
			s = next
			s.fromExit = false
			return s.filename(SaveName), nil, nil
		case F7:
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			s.mode = ExitSaveQuestion
			return s, nil, nil
		case Text, Insert, Delete, Backspace, Block, F1, Previous, Restore:
			return s.edit(event)
		case Reveal:
			if s.native == nil {
				return s, nil, refuse("Reveal requires retrieved document")
			}
			if s.selection {
				return s, nil, refuse("view change during block is unqualified")
			}
			if s.historyValid && s.history.Qualified() {
				h, err := s.history.InterruptGrouping()
				if err != nil {
					return s, nil, err
				}
				s.history = h
			}
			s.cursor = s.cursor.WithReveal(!s.cursor.Reveal())
			return s, nil, nil
		case Left, Right:
			if s.native == nil {
				return s, nil, refuse("cursor requires retrieved document")
			}
			direction := 1
			if event.Command == Left {
				direction = -1
			}
			cursor, err := s.cursor.Move(direction)
			if err != nil {
				return s, nil, err
			}
			if s.historyValid && s.history.Qualified() {
				var h document.History
				if event.Command == Right {
					h, err = s.history.Separate(document.RightBoundary)
				} else {
					h, err = s.history.InterruptGrouping()
				}
				if err != nil {
					return s, nil, err
				}
				s.history = h
			}
			s.cursor = cursor
			s.provenance = event.Command
			return s, nil, nil
		case Home:
			if s.native == nil || s.homes == 3 {
				return s, nil, refuse("Home prefix outside observed bounded sequence")
			}
			if h, err := s.interrupted(); err != nil {
				return s, nil, err
			} else {
				s.history = h
			}
			s.homes++
			return s, nil, nil
		case Up:
			if s.homes != 2 && s.homes != 3 {
				return s, nil, refuse("only Home2/3 Up endpoints observed")
			}
			gap := 0
			if s.native.TokenCount() > 0 {
				token, _ := s.native.Token(0)
				if token.Kind == document.Opaque {
					return s, nil, refuse("opaque initial code semantics unknown")
				}
				if s.homes == 2 && token.Kind == document.BoldOn {
					gap = 1
				}
			}
			cursor, err := s.native.CursorAt(gap)
			if err != nil {
				return s, nil, err
			}
			s.cursor = cursor.WithReveal(s.cursor.Reveal())
			s.homes = 0
			return s, nil, nil
		}
	}
	return s, nil, refuse(fmt.Sprintf("command %d in %s", event.Command, s.mode))
}

func (s Session) editName(event Event) (Session, *Intent, error) {
	switch event.Command {
	case Macro:
		if s.mode == SaveName {
			return s, nil, nil
		}
		return s, nil, refuse("macro nesting in this filename prompt unqualified")
	case F1:
		s.newMacro = nil
		s.mode = Editing
		s.fromExit = false
		s.notice = ""
		return s, nil, nil
	case Escape:
		if s.mode == RetrieveName {
			s.mode = Editing
			s.notice = ""
			return s, nil, nil
		}
	case Text:
		if len(event.Text) > storage.MaxNameBytes {
			return s, nil, refuse("filename draft safety bound")
		}
		if event.Text == "" {
			return s, nil, refuse("empty filename text")
		}
		for _, r := range event.Text {
			if r < 0x20 || r > 0x7e {
				return s, nil, refuse("filename requires explicit ASCII profile")
			}
		}
		draft, caret := s.draft, s.caret
		if !s.touched {
			s.editedPrefill = false
			draft = ""
			caret = 0
		}
		if len(draft)+len(event.Text) > storage.MaxNameBytes {
			return s, nil, refuse("filename draft safety bound")
		}
		s.draft = draft[:caret] + event.Text + draft[caret:]
		s.caret = caret + len(event.Text)
		s.touched = true
		return s, nil, nil
	case Right:
		if s.mode == RetrieveName || s.mode == MacroName {
			return s, nil, refuse("Retrieve filename editing unobserved")
		}
		if s.caret >= len(s.draft) {
			return s, nil, refuse("filename caret boundary")
		}
		if !s.touched && s.draft != "" {
			s.editedPrefill = true
		}
		s.caret++
		s.touched = true
		return s, nil, nil
	case Backspace:
		if s.mode == RetrieveName || s.mode == MacroName {
			return s, nil, refuse("Retrieve filename editing unobserved")
		}
		if s.caret == 0 {
			return s, nil, refuse("filename caret boundary")
		}
		s.draft = s.draft[:s.caret-1] + s.draft[s.caret:]
		s.caret--
		s.touched = true
		return s, nil, nil
	case Enter:
		if s.mode == SaveName && s.editedPrefill {
			return s, nil, refuse("prefilled filename edit has only cancel evidence")
		}
		if s.fromExit {
			return s, nil, refuse("save-on-exit completion not observed; F1 can return to document")
		}
		resolve := s.resolve
		if s.mode == MacroName {
			resolve = s.resolveMacro
		}
		dir, name, err := resolve(s.draft)
		if err != nil {
			return s, nil, err
		}
		if s.mode == MacroName {
			return s.begin(ReadMacro, dir, name)
		}
		if s.mode == RetrieveName {
			return s.begin(Read, dir, name)
		}
		if s.source == nil && s.newPhase != NewUnpublished {
			return s, nil, refuse("new document writer unavailable")
		}
		s.lastSave = NotSaved
		return s.begin(Observe, dir, name)
	}
	return s, nil, refuse("unobserved filename command")
}

func receipt(result Result, applied bool) Receipt {
	result = copyResult(result)
	record := Receipt{Result: result, Applied: applied, DestinationMayExist: (!applied || result.Err != nil) && result.Outcome.State == storage.Published}
	var failure *storage.Failure
	if errors.As(result.Err, &failure) {
		copy := *failure
		record.Result.Err = &copy
		// A typed storage failure is authoritative about its transaction stage.
		record.DestinationMayExist = failure.DestinationMayExist
		record.Candidate = failure.Candidate
	}
	return record
}
func (s Session) complete(result Result) (Session, *Intent, error) {
	result = copyResult(result)
	if s.pending != nil && s.pending.firstProof != nil {
		if result.Operation != s.pending.Operation || result.Revision != s.pending.Revision || result.Kind != s.pending.Kind {
			s.detached = receipt(result, false)
			return s, nil, nil
		}
		return s.completeFirst(result)
	}
	if s.pending == nil || result.Operation != s.pending.Operation || result.Revision != s.pending.Revision || result.Kind != s.pending.Kind {
		if s.newMacro != nil && s.pending != nil && s.pending.Kind == ReadMacro {
			return s, nil, refuse("foreign New macro root result")
		}
		s = s.advanced()
		s.newMacro = nil
		s.detached = receipt(result, false)
		return s, nil, nil
	}
	request := *s.pending
	if request.Kind == ReadMacro {
		if s.newMacro != nil || (request.macroProof != nil && request.macroProof.target != nil) {
			if !s.validNewMacroRead(&request) {
				return s, nil, refuse("New macro root is stale or foreign")
			}
		}
		var err error
		result, err = acceptedMacroResult(&request, result, s.bufferID)
		if err != nil {
			return s, nil, err
		}
	}
	if result.Revision != s.revision {
		s.pending = nil
		s.mode = Editing
		s = s.advanced()
		s.detached = receipt(result, false)
		s.notice = "Older revision completed; current edits and name retained"
		return s, nil, nil
	}
	s = s.advanced()
	s.pending = nil
	s.mode = s.resume
	s.last = receipt(result, true)
	if result.Err != nil {
		s.notice = result.Err.Error()
		if request.Kind == ReadMacro {
			s.newMacro = nil
			s.mode = Editing
			s.notice = "Macro acquisition failed: " + result.Err.Error()
			if result.macroMissing {
				display := s.draft
				if !strings.HasSuffix(display, ".WPM") {
					display += ".WPM"
				}
				s.mode = MacroError
				s.notice = "ERROR: File not found -- " + display
			}
		}
		if request.Kind == Read && errors.Is(result.Err, fs.ErrNotExist) {
			s.notice = "ERROR: File not found -- " + s.draft
			s.touched = false

// Safety policies, not original application limits. The gap index is at most
// (MaxTokens+1)*4 bytes; a presentation allocates at most MaxWindowTokens elements.
const (
	MaxTokens       = 1 << 20
	MaxWindowTokens = 4096
)

type Category string

const (
	InvalidDocument  Category = "InvalidDocument"
	InvalidPosition  Category = "InvalidPosition"
	InvalidDirection Category = "InvalidDirection"
	TokenLimit       Category = "TokenLimit"
	WindowLimit      Category = "WindowLimit"
	UnsupportedToken Category = "UnsupportedToken"

// Failure identifies a document refusal. Native-reader errors pass through with
// their original *wp51.Failure type. A negative index/span denotes no token.
type Failure struct {
	Category   Category
	TokenIndex int
	Span       wp51.Span
}

func (e *Failure) Error() string {
	return fmt.Sprintf("document: %s at token %d, span %d+%d", e.Category, e.TokenIndex, e.Span.Offset, e.Span.Length)
}
func failure(k Category, index int, span wp51.Span) error { return &Failure{k, index, span} }

type Kind uint8

	ASCII
	BoldOn
	BoldOff

// Token is an independent inspection value. Raw always retains the exact native
// bytes, even for an opaque token whose movement/presentation is unsupported.

	Kind Kind
	Span wp51.Span

}

// Document must be constructed by Parse or New. It retains one immutable dependency
// image and a compact private absolute-gap index, with no mutable document state.
type Document struct{ state *revisionState }

// A caller may copy or replace the exported Document wrapper. Cursor and
// Revision capture this private immutable state, never that mutable wrapper.
// No state pointer, image pointer or gap slice is exposed to callers.
type revisionState struct {
	image *wp51.Image
	gaps  []uint32
	birth *birthLineage
}

func (s *revisionState) valid() bool {
	return s != nil && s.image != nil && s.birth != nil && len(s.gaps) > 0
}

// Parse delegates all native validity decisions to the approved reader. The
// following indexing pass only locates already-validated token spans; it does not
// decode opaque payloads, nested text, paired-code relationships or edit safety.
func Parse(input []byte) (*Document, error) { return parseRevision(input, nil) }

// One native parser/indexer. A nil birth means an explicit Loaded acquisition;
// internal validation and New supply their captured immutable lineage.
func parseRevision(input []byte, birth *birthLineage) (*Document, error) {
	image, err := wp51.Parse(input)
	if err != nil {
		return nil, err
	}
	info := image.Inspect()
	if info.OuterTokenCount > MaxTokens {
		return nil, failure(TokenLimit, -1, info.Body)
	}
	body, _ := image.Bytes(info.Body)
	gaps := make([]uint32, info.OuterTokenCount+1)
	offset := 0
	for i := 0; i < info.OuterTokenCount; i++ {
		gaps[i] = uint32(info.Body.Offset + offset)
		offset += framedWidth(body[offset:])
	}
	gaps[len(gaps)-1] = uint32(info.Body.Offset + len(body))
	if birth == nil {
		birth = &birthLineage{kind: loadedBirth, originalImage: image}
	}
	return &Document{state: &revisionState{image: image, gaps: gaps, birth: birth}}, nil
}

// These SDK WP51 outer-width facts are used only after wp51.Parse has checked
// every delimiter/length/bound. Reusing their span lengths is not another native
// validator or permission to interpret payloads. In particular C0 remains opaque.
func framedWidth(raw []byte) int {
	lead := raw[0]
	if lead < 0xc0 {
		return 1
	}
	if lead < 0xd0 {
		return [...]int{4, 9, 11, 3, 3, 5, 6, 7, 4, 5, 6, 6, 8, 10, 10, 12}[lead-0xc0]
	}
	return int(binary.LittleEndian.Uint16(raw[2:4])) + 4
}
func identify(raw []byte) Kind {
	if len(raw) == 1 && raw[0] >= 0x20 && raw[0] <= 0x7e {
		return ASCII

	if len(raw) == 3 && raw[1] == 0x0c && raw[0] == raw[2] {
		if raw[0] == 0xc3 {
			return BoldOn
		}
		if raw[0] == 0xc4 {
			return BoldOff
		}
	}
	return Opaque
}

// the caller's previous cursor on failure. Its document remains immutable.

// CursorAt constructs a valid token-gap coordinate, not an original GoTo/Home
// command. Zero and TokenCount are the body boundaries, never byte positions.
func (d *Document) CursorAt(gap int) (Cursor, error) {

		return Cursor{}, failure(InvalidDocument, -1, noSpan)
	}

		return Cursor{}, failure(InvalidPosition, gap, noSpan)

		return 0, failure(InvalidDocument, -1, noSpan)
	}
	return int(c.state.gaps[c.gap]), nil

func (c Cursor) Move(direction int) (Cursor, error) {
	d := &Document{state: c.state}
	if !c.state.valid() {
		return c, failure(InvalidDocument, -1, noSpan)
	}
	if direction != -1 && direction != 1 {
		return c, failure(InvalidDirection, c.gap, noSpan)
	}
	target := c.gap + direction
	if target < 0 || target > d.TokenCount() {
		return c, failure(InvalidPosition, target, noSpan)
	}
	index := c.gap
	if direction < 0 {
		index--
	}
	token, _ := d.Token(index)
	if token.Kind == Opaque {
		return c, failure(UnsupportedToken, index, token.Span)
	}
	c.gap = target
	return c, nil
}

// Element preserves source/token identity in both views, including the empty
// normal text of Bold codes. Text is derived display, never native serialization.
type Element struct {
	TokenIndex int
	Kind       Kind
	Span       wp51.Span
	Text       string
}
type Presentation struct {
	StartGap, EndGap, CursorGap int
	Reveal                      bool
	Elements                    []Element
}

// Present renders a bounded token window, refusing an opaque token before
// returning any elements. It does not wrap, count terminal cells or infer layout.
func (c Cursor) Present(start, count int) (Presentation, error) {
	d := &Document{state: c.state}
	if !d.valid() {
		return Presentation{}, failure(InvalidDocument, -1, noSpan)
	}
	if start < 0 || start > d.TokenCount() || count < 0 || count > d.TokenCount()-start {
		return Presentation{}, failure(InvalidPosition, start, noSpan)
	}
	if count > MaxWindowTokens {
		return Presentation{}, failure(WindowLimit, start, noSpan)
	}
	// Acquire only the requested contiguous window, not the full body. Indexes
	// already identify valid complete-token spans, so this remains input bounded.
	window := wp51.Span{Offset: int(d.state.gaps[start]), Length: int(d.state.gaps[start+count] - d.state.gaps[start])}
	raw, _ := d.state.image.Bytes(window)
	for i := start; i < start+count; i++ {
		span := d.span(i)
		local := span.Offset - window.Offset
		if identify(raw[local:local+span.Length]) == Opaque {
			return Presentation{}, failure(UnsupportedToken, i, span)
		}
	}
	p := Presentation{StartGap: start, EndGap: start + count, CursorGap: c.gap, Reveal: c.reveal, Elements: make([]Element, count)}
	for i := start; i < start+count; i++ {
		span := d.span(i)
		local := span.Offset - window.Offset
		token := raw[local : local+span.Length]
		kind := identify(token)
		text := ""
		if kind == ASCII {
			text = string(token)
		} else if c.reveal {
			if kind == BoldOn {
				text = "[BOLD]"
			} else {
				text = "[bold]"
			}
		}
		p.Elements[i-start] = Element{TokenIndex: i, Kind: kind, Span: span, Text: text}
	}
	return p, nil
}

func (d *Document) editASCII(c Cursor, text []byte, over bool) (EditResult, error) {
	unchanged := EditResult{Document: d, Cursor: c}
	if !d.valid() {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if c.state != d.state {
		return unchanged, failure(RevisionMismatch, c.gap, noSpan)
	}
	// Cursor fields are private, but retain explicit checks before indexing.
	if c.gap < 0 || c.gap > d.TokenCount() {
		return unchanged, failure(InvalidPosition, c.gap, noSpan)
	}
	_, e := d.mutationAdmission()
	if e != nil {
		return unchanged, e
	}
	if len(text) > MaxMutationTokens {
		return unchanged, failure(MutationLimit, c.gap, noSpan)
	}
	for i, b := range text {
		if b < 0x20 || b > 0x7e {
			return unchanged, failure(InvalidASCII, i, noSpan)
		}
	}
	if len(text) == 0 {
		unchanged.receipt = &asciiEditReceipt{before: d.state, after: d.state, cursor: c}
		return unchanged, nil
	}
	raw := d.state.image.Export()
	start := int(d.state.gaps[c.gap])
	end := start
	replaced := 0
	if over {
		for replaced < len(text) && c.gap+replaced < d.TokenCount() {
			span := d.span(c.gap + replaced)
			if identify(raw[span.Offset:span.Offset+span.Length]) != ASCII {
				break
			}
			end += span.Length
			replaced++
		}
	}
	// Subtraction precedes addition, with input text already bounded.
	size := len(raw) - (end - start) + len(text)
	count := d.TokenCount() - replaced + len(text)
	if size > MaxMutationBytes || count > MaxMutationTokens {
		return unchanged, failure(MutationLimit, c.gap, noSpan)
	}
	bodyStart := int(d.state.gaps[0])
	body := make([]byte, 0, size-bodyStart)
	body = append(body, raw[bodyStart:start]...)
	body = append(body, text...)
	body = append(body, raw[end:]...)
	r, e := d.publishBody(c, body, count, c.gap+len(text), false)
	if e == nil {
		r.receipt = &asciiEditReceipt{before: d.state, after: r.Document.state, cursor: r.Cursor, changed: r.Changed, replaced: replaced, inputLength: len(text)}
	}
	return r, e
}

// The sole semantic candidate publisher. Only domain commands call it, after
// whole-token edits. forceFormat is authorized for confirmed blocked-pair repair;
// no-edit/decline/preview paths never call it. It preserves the owned input prefix
// and applies only checked indexed packet6 bit3, never a template or relocation.
func (d *Document) publishBody(c Cursor, body []byte, tokens, gap int, forceFormat bool) (EditResult, error) {
	unchanged := EditResult{Document: d, Cursor: c}
	if !d.valid() {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if c.state != d.state {
		return unchanged, failure(RevisionMismatch, c.gap, noSpan)
	}
	flags, e := d.mutationAdmission()
	if e != nil {
		return unchanged, e
	}
	start := int(d.state.gaps[0])
	if tokens < 0 || tokens > MaxMutationTokens || len(body) > MaxMutationBytes-start {
		return unchanged, failure(MutationLimit, gap, noSpan)
	}
	if gap < 0 || gap > tokens {
		return unchanged, failure(InvalidPosition, gap, noSpan)
	}
	raw := d.state.image.Export()
	changed := !bytes.Equal(raw[start:], body)
	cursor := c
	cursor.gap = gap
	if !changed && (!forceFormat || raw[flags]&8 != 0) {
		return EditResult{Document: d, Cursor: cursor}, nil
	}
	candidate := make([]byte, 0, start+len(body))
	candidate = append(candidate, raw[:start]...)
	candidate = append(candidate, body...)
	candidate[flags] |= 8
	next, e := parseRevision(candidate, d.state.birth)
	if e != nil {
		return unchanged, e
	}
	if next.TokenCount() != tokens {
		return unchanged, failure(InvalidDocument, -1, noSpan)
	}
	if e = next.CheckMutation(); e != nil {
		return unchanged, e
	}
	cursor.state = next.state
	return EditResult{Document: next, Cursor: cursor, Changed: changed}, nil
}

func (d *Document) mutationAdmission() (int, error) {

		return 0, failure(InvalidDocument, -1, noSpan)
	}
	info := d.state.image.Inspect()
	if info.Prefix.Length+info.Body.Length > MaxMutationBytes || d.TokenCount() > MaxMutationTokens {
		return 0, failure(MutationLimit, -1, info.Body)
	}
	if len(info.Blocks) != 2 || len(info.Packets) != 5 {
		return 0, failure(UnsupportedPrefix, -1, info.Prefix)
	}
	raw := d.state.image.Export()
	// The approved reader checks native identity, arithmetic, bounds, cycles and
	// delimiters. Consume its actual packet/block spans, never reparse a codec.
	first, last := info.Blocks[0], info.Blocks[1]
	if first.Data.Offset != 16 || first.Data.Length != 50 || last.Data.Length != 50 || first.Count != 5 || last.Count != 5 || first.Next != uint32(last.Data.Offset) || last.Next != 0 {
		return 0, failure(UnsupportedPrefix, -1, info.Prefix)
	}
	// Reader deliberately ignores unused slots/terminator payload; mutation may not.
	if !bytes.Equal(raw[last.Data.Offset+20:last.Data.Offset+50], make([]byte, 30)) {
		return 0, failure(UnsupportedPrefix, -1, last.Data)
	}

	// linked index order. Physical payload offsets still come from Inspect.

		types = [...]uint16{6, 7, 15, 12, 8}

			return 0, failure(UnsupportedPrefix, -1, p.Index)

		return h, failure(MutationLimit, -1, noSpan)
	}
	next := *h.state
	next.document = after
	next.seedStage = seedUnqualified
	if s.kind == CharacterDeletion && s.target == 0 && h.state.count == 0 && len(f.kinds) == 1 && f.kinds[0] == ASCII && wp40Shape(after) {
		next.seedStage = seedAwaitingBrowse
		next.seedView = s.cursor.reveal
	}
	next.groupUnknown = false
	consecutive := s.kind == CharacterDeletion && s.direction == ForwardDelete && next.groupOpen && s.cursor.gap == next.groupGap && next.count > 0 && next.entries[0].kind == CharacterDeletion
	if consecutive {
		old := next.entries[0]
		if len(old.fragment.native) > MaxHistoryFragmentBytes-len(f.native) || len(old.fragment.kinds) > MaxMutationTokens-len(f.kinds) {
			return h, failure(MutationLimit, -1, noSpan)
		}
		old.fragment = joinFragments(old.fragment, f)
		old.commands++
		next.entries[0] = old
	} else {
		for i := HistoryCapacity - 1; i > 0; i-- {
			next.entries[i] = next.entries[i-1]
		}
		next.entries[0] = historyEntry{f, s.kind, 1}
		if next.count < HistoryCapacity {
			next.count++
		}
	}
	next.groupOpen = s.kind == CharacterDeletion && s.direction == ForwardDelete

		}

		next.steps++

		return h, failure(HistoryEmpty, -1, noSpan)

		return h, failure(HistoryMismatch, -1, noSpan)
	}
	next := *h.state
	if next.observed != nil {
		if p.state.steps != 3 {
			return h, failure(UnsupportedHistoryEffect, -1, noSpan)
		}
		observation := *next.observed
		observation.browsed = true
		next.observed = &observation
	}
	if next.seedStage == seedAwaitingBrowse && p.state.steps == 3 && p.state.cursor.gap == 0 && p.state.cursor.reveal == next.seedView {
		next.seedStage = seedBrowsed
	} else if next.seedStage != seedRightSeparated {
		next.seedStage = seedUnqualified
	}
	// No post-preview continued-delete grouping probe exists. Preserve entries;
	// refuse that unqualified interleaving until a known Save/Right separator.
	next.groupUnknown = next.groupUnknown || next.groupOpen
	next.groupOpen = false
	return History{&next}, nil
}
func (d *Document) Restore(h History, p Preview) (Deletion, error) {
	unchanged := unchangedDeletion(d, p.Cursor(), h)
	if !p.Active() {
		return unchanged, failure(HistoryEmpty, -1, noSpan)
	}
	if e := d.checkDeletion(p.Cursor(), h); e != nil {
		return unchanged, e
	}
	if p.state.history != h.state {
		return unchanged, failure(HistoryMismatch, -1, noSpan)
	}
	if h.state.observed != nil && (h.state.observed.restored || !h.state.observed.browsed || p.state.index != 0) {
		return unchanged, failure(UnsupportedHistoryEffect, -1, noSpan)
	}
	f := h.state.entries[p.fragmentIndex()].fragment
	if len(f.kinds) > MaxMutationTokens-d.TokenCount() {
		return unchanged, failure(MutationLimit, p.Cursor().gap, noSpan)
	}
	raw := d.state.image.Export()
	start := int(d.state.gaps[0])
	at := int(d.state.gaps[p.Cursor().gap])
	bodyLength := len(raw) - start
	if len(f.native) > MaxMutationBytes-start-bodyLength {
		return unchanged, failure(MutationLimit, p.Cursor().gap, noSpan)
	}
	body := make([]byte, 0, bodyLength+len(f.native))
	body = append(body, raw[start:at]...)
	body = append(body, f.native...)
	body = append(body, raw[at:]...)
	edit, e := d.publishBody(p.Cursor(), body, d.TokenCount()+len(f.kinds), p.Cursor().gap+len(f.kinds), false)
	if e != nil {
		return unchanged, e
	}
	next := *h.state
	next.document = edit.Document.state
	if next.observed != nil {
		observation := *next.observed
		observation.restored = true
		observation.gap = edit.Cursor.gap
		observation.browsed = false
		next.observed = &observation
	}
	next.groupOpen = false
	next.groupUnknown = false
	// EC204 proves one attributed Restore, not its later stack lifetime. Keep
	// prior bytes but refuse subsequent queries; plain/opening-repair retention
	// is independently observed and remains enabled.
	if h.state.entries[p.fragmentIndex()].kind == AttributedDeletion {
		next.effectUnqualified = true
	}
	return Deletion{&deletionState{before: d.state, after: edit.Document.state, beforeHistory: h.state, afterHistory: &next, cursor: edit.Cursor, kind: Restoration, applied: true, bodyChanged: edit.Changed, nativeChanged: d.state != edit.Document.state}}, nil
}

	Kind       Kind

	Text       string
}

	Elements                                           []OverlayElement
	FragmentStart, FragmentEnd, CursorGap, TotalTokens int

}

func (p Preview) Present(start, count int, reveal bool) (Overlay, error) {
	if !p.Active() {
		return Overlay{}, failure(HistoryEmpty, -1, noSpan)
	}
	d := &Document{state: p.state.history.document}
	f := p.state.history.entries[p.fragmentIndex()].fragment
	gap := p.state.cursor.gap
	total := d.TokenCount() + len(f.kinds)
	if start < 0 || start > total || count < 0 || count > total-start {
		return Overlay{}, failure(InvalidPosition, start, noSpan)
	}
	if count > MaxWindowTokens {
		return Overlay{}, failure(WindowLimit, start, noSpan)
	}
	out := Overlay{Elements: make([]OverlayElement, count), FragmentStart: gap, FragmentEnd: gap + len(f.kinds), CursorGap: gap, TotalTokens: total, Reveal: reveal}
	for i := start; i < start+count; i++ {
		var e OverlayElement
		if i >= gap && i < gap+len(f.kinds) {
			index := i - gap
			e = OverlayElement{Origin: DeletionFragment, TokenIndex: index, Kind: f.kinds[index], Span: noSpan, Raw: []byte(f.native[f.offset(index):f.offset(index+1)])}
		} else {
			index := i
			if i >= gap+len(f.kinds) {
				index -= len(f.kinds)
			}
			token, _ := d.Token(index)
			e = OverlayElement{Origin: DocumentToken, TokenIndex: index, Kind: token.Kind, Span: token.Span, Raw: token.Raw}
		}
		if e.Kind == ASCII {
			e.Text = string(e.Raw)
		} else if reveal {
			if e.Kind == BoldOn {
				e.Text = "[BOLD]"
			} else {
				e.Text = "[bold]"
			}
		}
		out.Elements[i-start] = e
	}
	return out, nil
}

## State and position notation

One authoritative document state must serve normal and Reveal Codes views.
Use token gaps to express semantic cursor positions; a token can be an original
character identity or a complete native code. Never put the cursor inside a
C3/C4 byte sequence, or equate terminal cells, displayed inches and byte offsets.
`|` below is a gap, not a stored character. Selection `[start,end)` is a range
between gaps, including complete codes. The following notation is an
implementation contract derived from fixture framing, not an original internal
WP data structure:

- D: owned native image plus interpreted tokens and opaque spans; unrelated bytes
  and relationships remain preserved under FC-003/004.
- P: active token gap; R: Reveal Codes visible/hidden; M: insertion/typeover.
- B: no block or a block anchor/current gap; Q: normal/Delete Block/Undelete
  interaction context. F1 is context-dependent.
- H: remembered **deletion entries with command provenance**, newest first;
  h: optional preview choice. Preview is not an inserted document fragment.

The saved A fixture `A[BOLD]BC[bold]D` has B=324 and body
`41 C3 0C C3 42 43 C4 0C C4 44`. Its gaps g0..g6 correspond to absolute byte
offsets `324,325,328,329,330,333,334`. g1 is before Bold On, g2 after it,
g4 before Bold Off and g5 after it. The 3-byte codes are each one token here.
For the unsaved E fixture `A[BOLD]B[bold]C`, use the analogous token gaps without
claiming its missing pre-edit native file was captured. Printed code labels are
views; `[BOLD]` is not literal document text.

Movement setup in E/A is `Home,Home,Home,Up` followed by the exact Right sequence
in the archived recipe. g1 in A has displayed Pos 1.1 inches; g4 has Pos 1.3.

Center's after-code position is 4.1 inches in CMD51-051; removing it returns to
1 inch. A code can affect geometry without adding an ordinary character.

def sha(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


class Probe:
    def __init__(self, q, out):
        self.q, self.out, self.trace, self.checks = q, out, [], []

    def key(self, keys):
        self.trace.append({"key": keys})
        for key in keys.split(","):
            self.q.key(key.strip())

    def type(self, text):
        self.trace.append({"type": text})
        self.q.type(text)

    def capture(self, label):
        self.trace.append({"capture": label})
        return self.q.capture(self.out / label)

    def check(self, name, condition):
        self.checks.append({"check": name, "passed": bool(condition)})
        if not condition:
            raise AssertionError(name)


def menus(p):
    p.capture("before")
    p.key("alt-equal")
    p.q.wait_for("File Edit Search")
    s = p.capture("menu-on")
    p.check("Menu bar appears", "File Edit Search Layout" in s)
    p.key("alt-equal")
    p.q.wait_absent("File Edit Search")
    s = p.capture("menu-off")
    p.check("Menu bar disappears", "File Edit Search" not in s)
    p.check("Visible position retained", 'Doc 1 Pg 1 Ln 1" Pos 1"' in s)

                    if data[:4] != bytes.fromhex("ff575043") or not result["native_output"]["text_present"]:

    if any(r["result"] == "error" for r in run["results"]):

if __name__ == "__main__":
    main()
