def read(p):

    directory='/'.join(parts[:i+1]);need(directory in ALLOWED_DIRS,'unlisted directory create')
    os.mkdir(name,0o755,dir_fd=fd);effects.append(dict(kind='mkdir',path=directory,state='created'));os.fsync(fd)
