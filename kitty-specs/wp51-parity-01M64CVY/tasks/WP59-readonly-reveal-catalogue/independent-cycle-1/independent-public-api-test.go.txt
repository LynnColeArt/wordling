func TestIndependentPublicCatalogue(t *testing.T){body:=[]byte{0x0c,0x0a,0xc3,0x0c,0xc3,0xd0,1,8,0,0x0a,0x0c,0x55,0x55,8,0,1,0xd0};x:=corpus(t,body);rev:=x.Revision();raw,_:=x.Export();p,e:=d.InspectRevealCatalogue(x,rev,0,4);if e!=nil{t.Fatal(e)};want:=[]string{"[HPg]","[HRt]","[BOLD]",""};for i,v:=range p.Elements{tok,_:=x.Token(i);if v.Text!=want[i]||v.Kind!=tok.Kind||v.Span!=tok.Span||v.TokenIndex!=i{t.Fatal(i,v,tok)}};if !p.TrueEOF||p.Elements[0].Kind!=d.Opaque||p.Elements[3].Span.Length!=12{t.Fatal(p)};p.Elements[0].Text="mutated";again,_:=d.InspectRevealCatalogue(x,rev,0,1);if again.Elements[0].Text!="[HPg]"||again.TrueEOF{t.Fatal(again)};other:=corpus(t,body);if _,e=d.InspectRevealCatalogue(x,other.Revision(),0,1);e==nil{t.Fatal("equal-byte identity forged")};for _,r:=range[][2]int{{math.MaxInt,0},{0,math.MaxInt},{-1,0},{0,-1}}{if _,e=d.InspectRevealCatalogue(x,rev,r[0],r[1]);e==nil{t.Fatal(r)}};eof,e:=d.InspectRevealCatalogue(x,rev,4,0);if e!=nil||!eof.TrueEOF||len(eof.Elements)!=0{t.Fatal(eof,e)};c,_:=x.CursorAt(0);if _,e=c.Move(1);e==nil{t.Fatal("opaque movement admitted")};if _,e=c.Present(0,4);e==nil{t.Fatal("Present relaxed")};if e=x.CheckMutation();e==nil{t.Fatal("mutation admitted")};after,_:=x.Export();if !bytes.Equal(raw,after)||x.Revision()!=rev{t.Fatal("native mutated")}}
