from pathlib import Path
import json,collections,xml.etree.ElementTree as ET
p=Path('/home/lynn/projects/wordling/internal/editor/evidence/retrieve-italics');summary={}
files=[p/(s+'.stdout')for s in ['union-green','full-go-test','focused-race']]+list((p/'replays').glob('*.stdout'))
for f in files:
 ev=[]
 for line in f.read_text().splitlines():
  try:ev.append(json.loads(line))
  except json.JSONDecodeError:pass
 term=[x for x in ev if x.get('Test')and x.get('Action')in('pass','fail','skip')];names={(x['Package'],x['Test'])for x in term};leaf=[x for x in term if not any(pkg==x['Package']and n.startswith(x['Test']+'/')for pkg,n in names)];label=str(f.relative_to(p)).removesuffix('.stdout');summary[label]={'named_including_parents':dict(collections.Counter(x['Action']for x in term)),'leaf':dict(collections.Counter(x['Action']for x in leaf)),'skips':[{'package':x['Package'],'test':x['Test'],'output':''.join(y.get('Output','')for y in ev if y.get('Test')==x['Test'])}for x in term if x['Action']=='skip']}
 suite=ET.Element('testsuite',name=label,tests=str(len(term)),failures=str(sum(x['Action']=='fail'for x in term)),skipped=str(sum(x['Action']=='skip'for x in term)))
 for x in term:
  c=ET.SubElement(suite,'testcase',name=x['Test'],classname=x['Package']);output=''.join(y.get('Output','')for y in ev if y.get('Test')==x['Test']);ET.SubElement(c,'system-out').text=output
  if x['Action']=='fail':ET.SubElement(c,'failure').text=output
  if x['Action']=='skip':ET.SubElement(c,'skipped').text=output
 target=f.with_name(f.stem+'-junit.xml');ET.ElementTree(suite).write(target,encoding='utf-8',xml_declaration=True);target.write_text('\n'.join(line.rstrip()for line in target.read_text().splitlines())+'\n')
summary['commands']={str(f.relative_to(p)):json.loads(f.read_text())for f in p.rglob('*.run.json')};summary['official_initial_baseline_junit']=ET.parse(p/'baseline-junit.xml').getroot().attrib
(p/'verification-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
