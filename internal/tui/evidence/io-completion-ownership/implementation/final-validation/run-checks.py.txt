from pathlib import Path
import subprocess,json,hashlib,os,sys,time
root=Path('/home/lynn/projects/wordling');out=Path('/tmp/wordling-wp110-final-checks');out.mkdir(exist_ok=True)
group=sys.argv[1]
env=os.environ.copy();env.update(GOTOOLCHAIN='local',GOPROXY='off',GOSUMDB='off')
owned=['internal/tui/model.go','internal/tui/runtime.go','internal/tui/io_completion_ownership_test.go','internal/tui/model_test.go','internal/tui/editable_test.go','internal/tui/original_test.go']
pins={p:hashlib.sha256((root/p).read_bytes()).hexdigest()for p in owned}
go=Path(subprocess.check_output(['which','go'],text=True).strip()).resolve()
def run(name,argv,overrides=None):
 local=env.copy();local.update(overrides or {})
 meta={'argv':argv,'cwd':str(root),'git_HEAD_before':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'working_sources':pins,'go_version':subprocess.check_output(['go','version'],text=True).strip(),'go_executable':str(go),'go_executable_sha256':hashlib.sha256(go.read_bytes()).hexdigest(),'env_overrides':{k:local[k]for k in ['GOTOOLCHAIN','GOPROXY','GOSUMDB']+list(overrides or {})},'chronology':'Working source follows committed actual primary RED52f26fd0; Git HEAD is metadata parent, not a false claim that it already names these uncommitted production bytes.'}
 start=time.monotonic()
 with (out/(name+'.stdout')).open('w')as a,(out/(name+'.stderr')).open('w')as b:r=subprocess.run(argv,cwd=root,env=local,stdout=a,stderr=b)
 meta['exit_code']=r.returncode;meta['elapsed_seconds']=time.monotonic()-start
 for p,h in pins.items():assert hashlib.sha256((root/p).read_bytes()).hexdigest()==h,p
 for ext in ['stdout','stderr']:
  p=out/(name+'.'+ext);meta[ext]={'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
 (out/(name+'.run.json')).write_text(json.dumps(meta,indent=2)+'\n');print(name,'exit',r.returncode,flush=True);return r.returncode
if group=='full':
 for name,argv in [('focused',['go','test','-mod=readonly','-count=1','-timeout=90s','-json','./internal/tui','-run','^Test(OrdinaryCompletion|BareUncertainResultKeepsAuthenticPendingWork|ActualPublicationAndUnboundForgedErrorAreDistinct|CloseGatesQueuedIOAndRetainsStartedReceipts)']),('full-test',['go','test','-mod=readonly','-count=1','-timeout=90s','-json','./...']),('full-vet',['go','vet','./...']),('full-build',['go','build','./...']),('normal-review-adapter',['python3','scripts/review_checks.py','--junitxml='+str(out/'normal-review-adapter.junit.xml')])]:
  if run(name,argv):sys.exit(1)
elif group=='race':
 sys.exit(run('focused-race',['go','test','-race','-mod=readonly','-count=1','-timeout=180s','-json','./internal/tui','./internal/editor','./internal/storage']))
elif group=='windows':
 win={'GOOS':'windows','GOARCH':'amd64','CGO_ENABLED':'0'}
 if run('windows-production-closure',['go','list','-deps','-json','./cmd/wordling'],win):sys.exit(1)
 text=(out/'windows-production-closure.stdout').read_text();dec=json.JSONDecoder();records=[]

  text=text.lstrip();obj,n=dec.raw_decode(text);records.append(obj);text=text[n:]
 app=[p for p in records if p.get('Module',{}).get('Path')=='github.com/lynncoleart/wordling']

 sources={}
 for p in app:
  for field in ['GoFiles','CgoFiles','SFiles','EmbedFiles']:
   for f in p.get(field,[]):
    path=Path(p['Dir'])/f;rel=str(path.relative_to(root));sources[rel]={'bytes':path.stat().st_size,'sha256':hashlib.sha256(path.read_bytes()).hexdigest()}
 (out/'windows-production-source-pins.json').write_text(json.dumps({'packages':packages,'selected_source_files':sources,'package_count':len(packages),'source_count':len(sources),'native_execution':False},indent=2)+'\n')
 if run('windows-production-vet',['go','vet',*packages],win):sys.exit(1)
 exe=out/'wordling.exe'
 if run('windows-production-build',['go','build','-o',str(exe),'./cmd/wordling'],win):sys.exit(1)
 (out/'windows-binary.json').write_text(json.dumps({'bytes':exe.stat().st_size,'sha256':hashlib.sha256(exe.read_bytes()).hexdigest(),'removed_after_hash':True,'native_execution':False},indent=2)+'\n');exe.unlink()
 run('windows-broad-vet',['go','vet','./...'],win)
 run('windows-broad-build',['go','build','./...'],win)
else:raise SystemExit('unknown check group')
