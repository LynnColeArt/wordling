def sha(b):return hashlib.sha256(b).hexdigest()
def readlist(path):
 raw=path.read_text();d=json.JSONDecoder();pkgs=[];files=[]
 while raw.strip():
  x,n=d.raw_decode(raw.lstrip());raw=raw.lstrip()[n:];pkgs.append({'package':x['ImportPath'],'directory':x['Dir'],'module':x.get('Module'),'GoFiles':x.get('GoFiles',[]),'CgoFiles':x.get('CgoFiles',[]),'SFiles':x.get('SFiles',[]),'EmbedFiles':x.get('EmbedFiles',[])})

   p=pathlib.Path(x['Dir'])/name;b=p.read_bytes();files.append({'package':x['ImportPath'],'path':str(p),'bytes':len(b),'sha256':sha(b)})
 return {'packages':pkgs,'inputs':files}
