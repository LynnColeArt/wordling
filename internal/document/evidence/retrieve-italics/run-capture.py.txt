from pathlib import Path
import subprocess,sys,json,time,os
label=sys.argv[1]; cmd=sys.argv[2:]; p=Path('/home/lynn/projects/wordling/internal/document/evidence/retrieve-italics'); start=time.time()
r=subprocess.run(cmd,cwd='/home/lynn/projects/wordling',text=True,capture_output=True)
(p/(label+'.stdout')).write_text(r.stdout); (p/(label+'.stderr')).write_text(r.stderr)
(p/(label+'.run.json')).write_text(json.dumps({'argv':cmd,'exit':r.returncode,'seconds':time.time()-start,'GOOS':os.environ.get('GOOS'),'GOARCH':os.environ.get('GOARCH')},indent=2))
print(label,r.returncode,r.stdout[-1500:],r.stderr[-1000:]);sys.exit(r.returncode)
