def stamp(s):
    return (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)


def facts(s):
    return dict(mode=stat.S_IMODE(s.st_mode), uid=s.st_uid, gid=s.st_gid,
        dev=s.st_dev, ino=s.st_ino, size=s.st_size, mtime_ns=s.st_mtime_ns, ctime_ns=s.st_ctime_ns)


def directory(path):
    if not path.is_absolute() or '..' in path.parts:
        raise RuntimeError('FixedPathInvariant')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        for part in path.parts[1:]:
            old = os.stat(part, dir_fd=fd, follow_symlinks=False)
            if not stat.S_ISDIR(old.st_mode):
                raise RuntimeError('ParentType:' + str(path))
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=fd)
            if stamp(old) != stamp(os.fstat(child)):
                os.close(child)
                raise RuntimeError('ParentChanged:' + str(path))
            os.close(fd)
            fd = child
        return fd
    except BaseException:
        os.close(fd)
        raise
