def acceptable_nonzero(name,argv):
 # Never infer host containment from a failed assertion alone. Only the actual
 # retained source-bound causal/reap outcomes of a closed CLI can qualify it.
 if name not in ('derivation97','envelope42'):return False
 xml=OUT/('derivation.xml' if name=='derivation97' else 'envelope.xml')
 if not xml.exists():return False
 proofs=[]
 for case in ET.parse(xml).getroot():
  failure=case.find('failure')
  if failure is None:continue
  if not case.attrib['name'].startswith('test_cli.'):return False
  text=failure.text or ''
  try:
   literal=text.split(' : ',1)[1].split('\n\n',1)[0];data=ast.literal_eval(literal)
   outcome=data['outcome'];journal=data['journal']
   assert outcome['supervisor_reaped'] is True and outcome['cleanup']=='STOPPED'
   assert any(row.get('phase')=='OWNED' and row.get('startticks') for row in journal)
   assert any(row.get('phase')=='FAULTED' for row in journal)
   assert data['ok'] is False and outcome['first_fault']==data['error']
   proofs.append(dict(case=case.attrib['name'],supervisor_actual_reaped=True,causal_owned_birth_retained=True,source_cleanup='STOPPED',first_fault=data['error'],escape_absence_not_claimed=True))
  except (KeyError,ValueError,AssertionError,IndexError,SyntaxError):return False
 if not proofs:return False
 (OUT/(name+'-nonzero-containment.json')).write_text(json.dumps(dict(proofs=proofs,basis='Actual source-bound worker causal birth + public client kernel waitpid reap, fixed authored closed CLI; not scalar STOPPED alone and no historical PID contact'),indent=2)+'\n')
 return True
