@dataclasses.dataclass(frozen=True)
class ManagedPolicy:
    total_seconds: float = 120
    startup_seconds: float = 15
    readiness_seconds: float = 60
    session_seconds: float = 960
    command_seconds: float = 15
    stream_bytes: int = 1048576
    output_bytes: int = 16777216
    processes: int = 8
    rss_bytes: int = 268435456
    commands: int = 16
    polls: int = 100000

    def __post_init__(self):
        limits=(900,15,60,960,15,1048576,16777216,8,268435456,16,100000)
        for field,limit in zip(dataclasses.fields(self),limits):
            value=getattr(self,field.name)
            if (type(value) not in (int,float) or not math.isfinite(value)
                    or not 0<value<=limit
                    or (not field.name.endswith('seconds') and type(value) is not int)):
                raise engine.Failure('invalid-managed-policy:'+field.name)

    @classmethod
    def test_v1(cls,total_seconds=120):
        return cls(total_seconds=total_seconds)

class StartupFailure(engine.Failure):
    def __init__(self,code,observation):
        super().__init__(code)
        self.observation=observation


def _identity(value):
    if value is None:
        return None
    v=dict(value);v['exe_devino']=tuple(v['exe_devino']);v['argv']=tuple(v['argv'])
    return IdentityObservation(**v)


def _frozen(value):
    if isinstance(value,dict):
        return tuple((k,_frozen(v)) for k,v in sorted(value.items()))
    if isinstance(value,list):
        return tuple(_frozen(x) for x in value)
    return value


def _source_observation(row,generation):
    return artifacts.SourceObservation(row['path'],row['size'],row['sha256'],row['device'],
        row['inode'],row['mode'],row['nlink'],row['mtime_ns'],row['ctime_ns'],generation,row['epoch'])


def _partial(v):
    if not v:
        return PartialOutcome('ControlLost','UNKNOWN',None,None,(),'UNKNOWN',(),False)
    v=dict(v);v['last_identity']=_identity(v['last_identity'])
    v['signals_issued']=tuple(tuple(x) if isinstance(x,list) else x for x in v['signals_issued']);v['unusable_resources']=tuple(v['unusable_resources'])
    return PartialOutcome(**v)


def _source_check(full=True):
    package=sys.modules[__package__]
    root,dev,ino,digests=package._source_binding
    directory=engine._path(root)
    info=directory.stat()
    if directory.is_symlink() or (info.st_dev,info.st_ino)!=(dev,ino):
        raise engine.Failure('PathRetarget')
    for name,digest in digests.items():
        p=directory/name
        if p.is_symlink() or not stat.S_ISREG(p.stat().st_mode):
            raise engine.Failure('PathRetarget')
        if full and hashlib.sha256(p.read_bytes()).hexdigest()!=digest:
            raise engine.Failure('SourceChanged')


def _json(raw):
    value=engine.decode_json(raw)
    engine._json_value(value)
    return value


def _frame(raw):
    if len(raw)>65536:
        raise engine.Failure('ControlLost')
    return struct.pack('!I',len(raw))+raw

class Session:
    __slots__=('_pid','_pidfd','_read','_write','_pid_origin','_generation','_handles',
               '_lock','_closed','_fault','_messages','_wire','_started','_policy',
               '_observations','_workspace','_last_partial','_sequence')
    def __new__(cls,key=None):
        if key is not _SESSION_KEY:
            raise TypeError('opaque session')
        return super().__new__(cls)

    def _initialize(self,pid,pidfd,read,write,generation,policy,workspace):
        self._pid,self._pidfd,self._read,self._write=pid,pidfd,read,write
        self._pid_origin=os.getpid();self._generation=generation;self._handles={}
        self._lock=threading.Lock();self._closed=False;self._fault=None
        self._messages=self._wire=self._sequence=0;self._started=time.monotonic()
        self._policy=policy;self._observations={};self._workspace=workspace
        self._last_partial=None

    def _mint(self,record,kind=None):
        classes={'source':artifacts.SourceLease,'working':artifacts.WorkingLease,
                 'executable':artifacts.ExecutableLease,'process':ProcessLease,'channel':CheckedChannel}
        cls=classes[kind or record['kind']]
        obj=cls(artifacts._MINT,self,record['token'],record['epoch'])
        self._handles[id(obj)]=(obj,record['token'],record['epoch'],cls)
        self._observations[record['token']]=record.get('observation',{})
        if cls is ProcessLease:
            obs=record['observation']
            self._last_partial=dict(first_fault=None,launch_state='CURRENTLY_OWNED',
                                    known_pid=obs['pid'],last_identity=obs,signals_issued=[],
                                    cleanup='UNKNOWN',unusable_resources=[],journal_durable=True)
        return obj

    def _ref(self,obj,cls):
        record=self._handles.get(id(obj))
        if (record is None or record[0] is not obj or type(obj) is not cls
                or obj._owner is not self or obj._token!=record[1] or obj._epoch!=record[2]):
            self._local_fail('InvalidLease')
        return dict(token=record[1],epoch=record[2])

    def _local_fail(self,code):
        if self._fault is None:
            try:self._rpc('fault',dict(code=code),allow_fault=True)
            except StartupFailure:pass
            self._fault=code
        raise StartupFailure(self._fault,_partial(self._last_partial))

    def _observation(self,lease):
        self._ref(lease,type(lease))
        # Facts only, defensive tree copy, never a lease constructor.
        row=self._observations.get(lease._token,{})
        if type(lease) in (artifacts.SourceLease,artifacts.WorkingLease,artifacts.ExecutableLease):
            return _source_observation(dict(row),self._generation)
        return _frozen(row)

    def _transport(self,request,deadline):
        raw=json.dumps(request,separators=(',',':'),allow_nan=False).encode()
        engine._json_value(request)
        packet=_frame(raw);offset=0;received=bytearray();size=None
        while True:
            if time.monotonic()>=deadline:
                raise engine.Failure('ControlLost')
            if select.select([self._pidfd],[],[],0)[0]:
                raise engine.Failure('CleanupUnknown')
            reads=[self._read];writes=[self._write] if offset<len(packet) else []
            readable,writable,_=select.select(reads,writes,[],min(.01,deadline-time.monotonic()))
            if writable:
                try:n=os.write(self._write,packet[offset:offset+16384])
                except BlockingIOError:n=0
                if n<0:raise engine.Failure('ControlLost')
                offset+=n
            if readable:
                chunk=os.read(self._read,16384)
                if not chunk:raise engine.Failure('ControlLost')
                received.extend(chunk)
                if len(received)>65540:raise engine.Failure('ControlLost')
                if len(received)>=4 and size is None:
                    size=struct.unpack('!I',received[:4])[0]
                    if size>65536:raise engine.Failure('ControlLost')
                if size is not None and len(received)==size+4:
                    self._wire+=len(packet)+len(received)
                    if self._wire>67108864:raise engine.Failure('ControlLost')
                    return _json(bytes(received[4:]))
                if size is not None and len(received)>size+4:
                    raise engine.Failure('ControlLost')

    def _rpc(self,operation,args=None,allow_fault=False):
        if (os.getpid()!=self._pid_origin or self._closed
                or (self._fault and not allow_fault)):
            raise StartupFailure(self._fault or 'InvalidLease',_partial(self._last_partial))
        if not self._lock.acquire(False):
            self._fault=self._fault or 'InvalidLease'
            if self._write>=0:
                os.close(self._write);self._write=-1
            raise StartupFailure(self._fault,_partial(self._last_partial))
        try:
            _source_check()
            self._messages+=1;self._sequence+=1
            if self._messages>4096:raise engine.Failure('ControlLost')
            deadline=min(self._started+self._policy.session_seconds,time.monotonic()+65)
            response=self._transport(dict(generation=self._generation,sequence=self._sequence,
                                          operation=operation,args=args or {}),deadline)
            if response.get('sequence')!=self._sequence or response.get('generation')!=self._generation:
                raise engine.Failure('ControlLost')
            if not response.get('ok'):
                self._last_partial=response.get('partial');self._fault=response.get('error','ControlLost')
                failure=StartupFailure(self._fault,_partial(self._last_partial))
                mutation=response.get('mutation_result')
                if mutation is not None and mutation.get('status')=='UNUSABLE':
                    mutation=dict(mutation)
                    mutation['tool_identity']=_identity(mutation['tool_identity'])
                    mutation['actual_argv']=tuple(mutation['actual_argv'])
                    mutation['successor']=None
                    failure.result=MutationResult(**mutation)
                raise failure
            if self._fault and not allow_fault:
                raise StartupFailure(self._fault,_partial(self._last_partial))
            return response.get('value')
        except StartupFailure:
            raise
        except BaseException as error:
            self._fault=self._fault or (error.code if isinstance(error,engine.Failure) else 'ControlLost')
            if self._last_partial is not None:
                self._last_partial=dict(self._last_partial,first_fault=self._fault,cleanup='UNKNOWN',
                                        journal_durable=False)
            # Closing only our control write end causes independent worker containment.
            if self._write>=0:
                os.close(self._write);self._write=-1
            raise StartupFailure(self._fault,_partial(self._last_partial)) from error
        finally:
            self._lock.release()

    def capture(self,request):
        if type(request) is not artifacts.SourceRequest:self._local_fail('SourceMismatch')
        return self._mint(self._rpc('capture',dataclasses.asdict(request)))

    def prepare(self,source,basename):
        return self._mint(self._rpc('prepare',dict(source=self._ref(source,artifacts.SourceLease),basename=basename)))

    def tool(self,source):
        return self._mint(self._rpc('tool',dict(source=self._ref(source,artifacts.SourceLease))))

    def _spec(self,spec,cls):
        if type(spec) is not cls or type(spec.argv) is not tuple or type(spec.env) is not tuple:
            self._local_fail('InvalidLease')
        argv=[]
        for a in spec.argv:
            if type(a) is str:argv.append(a)
            elif type(a) is artifacts.ResourceArgument:
                if type(a.prefix) is not str or type(a.suffix) is not str:self._local_fail('InvalidLease')
                if type(a.lease) not in (artifacts.SourceLease,artifacts.WorkingLease):self._local_fail('InvalidLease')
                argv.append(dict(prefix=a.prefix,ref=self._ref(a.lease,type(a.lease)),suffix=a.suffix))
            else:self._local_fail('InvalidLease')
        return dict(tool=self._ref(spec.tool,artifacts.ExecutableLease),argv=argv,env=[list(x) for x in spec.env])

    def mutate(self,target,spec):
        value=self._spec(spec,artifacts.MutationSpec)
        value['target']=self._ref(target,artifacts.WorkingLease)
        if type(spec.immutable_inputs) is not tuple:self._local_fail('InvalidLease')
        value['inputs']=[self._ref(x,artifacts.SourceLease) for x in spec.immutable_inputs]
        v=self._rpc('mutate',value)
        # Old working handle becomes stale; result alone is not its successor.
        self._observations[target._token]=v['resource']['observation']
        successor=self._mint(v['resource'])
        result=v['result'];result['tool_identity']=_identity(result['tool_identity'])
        result['actual_argv']=tuple(result['actual_argv'])
        return MutationResult(**result,successor=successor)

    def launch(self,spec,resources=()):
        if type(resources) is not tuple:self._local_fail('InvalidLease')
        args=self._spec(spec,artifacts.LaunchSpec)
        args['resources']=[self._ref(x,artifacts.WorkingLease) for x in resources]
        return self._mint(self._rpc('launch',args))

    def snapshot(self,target):
        v=self._rpc('snapshot',dict(target=self._ref(target,artifacts.WorkingLease)))
        row=v['snapshot']
        self._observations[target._token]=v['resource']['observation']
        successor=target if v['resource']['epoch']==target._epoch else self._mint(v['resource'])
        return Snapshot(v['resource']['token'],row['epoch'],row['size'],row['sha256'],
                        (row['device'],row['inode'],row['mode']),True,successor=successor)

    def _process_identity(self,lease):
        return _identity(self._rpc('identity',dict(process=self._ref(lease,ProcessLease))))

    def _endpoint(self,lease,basename):
        return self._mint(self._rpc('endpoint',dict(process=self._ref(lease,ProcessLease),basename=basename)))

    def _exchange(self,channel,request,reply,timeout):
        if type(request) is not bytes or type(reply) is not bytes:self._local_fail('InvalidLease')
        v=self._rpc('exchange',dict(channel=self._ref(channel,CheckedChannel),request=request.hex(),
                                    reply=reply.hex(),timeout=timeout))
        v['current_identity']=_identity(v['current_identity']);return ReadyObservation(**v)

    def _send(self,channel,raw,timeout):
        if type(raw) is not bytes:self._local_fail('InvalidLease')
        return IOObservation(**self._rpc('send',dict(channel=self._ref(channel,CheckedChannel),
                                                   raw=raw.hex(),timeout=timeout)))

    def _receive(self,channel,max_bytes,timeout):
        return bytes.fromhex(self._rpc('receive',dict(channel=self._ref(channel,CheckedChannel),
                                                      max_bytes=max_bytes,timeout=timeout)))

    def _channel_close(self,channel):
        self._rpc('channel_close',dict(channel=self._ref(channel,CheckedChannel)))

    def _finish(self,process,timeout):
        v=self._rpc('finish',dict(process=self._ref(process,ProcessLease),timeout=timeout))
        v['last_verified_identity']=_identity(v['last_verified_identity'])
        v['signals_issued']=tuple(v['signals_issued']);v['owned_session_remaining']=tuple(v['owned_session_remaining'])
        return ExitObservation(**v)

    def _cancel(self,process):
        self._ref(process,ProcessLease)
        try:self._rpc('cancel')
        except StartupFailure as error:return error.observation
        self._local_fail('Cancelled')

    def close(self):
        global _ACTIVE_SESSION
        if self._closed:
            raise StartupFailure('InvalidLease',_partial(self._last_partial))
        value=None
        if os.getpid()!=self._pid_origin:
            raise StartupFailure('InvalidLease',_partial(self._last_partial))
        try:
            if self._write>=0:
                try:value=self._rpc('close',allow_fault=True)
                except StartupFailure:pass
        finally:
            self._closed=True
            if _ACTIVE_SESSION is self:_ACTIVE_SESSION=None
            for name in ('_read','_write'):
                fd=getattr(self,name)
                if fd>=0:os.close(fd);setattr(self,name,-1)
            until=time.monotonic()+1.3;reaped=False
            while time.monotonic()<until:
                result=os.waitpid(self._pid,os.WNOHANG)
                if result[0]:reaped=True;break
                time.sleep(.01)
            os.close(self._pidfd)
        if value is None:
            return SessionOutcome('CLOSED',self._fault,0,(),(),(),time.monotonic()-self._started,
                                  reaped,'UNKNOWN')
        return SessionOutcome(value['terminal_state'],value['first_fault'],value['charged_operations'],
                              _frozen(value['spawn_observations']),
                              tuple(_source_observation(r,self._generation) for r in value['resources']),
                              tuple(tuple(x) for x in value['counters']),value['elapsed'],reaped,value['cleanup'])


class _Worker:
    """The only managed foreground parent/reaper; serial, continuously monitored."""
    def __init__(self,directory,fd,journal,read,write,parent,parent_fd,generation,stream,managed):
        self.directory,self.fd,self.journal=directory,fd,journal
        self.read,self.write,self.parent,self.parent_fd=read,write,parent,parent_fd
        self.generation,self.stream,self.policy=generation,stream,managed
        self.root_stamp=engine._stamp(os.fstat(fd))[:3]
        self.journal_stamp=engine._stamp(os.fstat(journal))[:3]
        self.started=time.monotonic();self.first_fault=None;self.durable=True
        self.state='PREPARING';self.events=1;self.journal_bytes=os.fstat(journal).st_size
        self.polls=self.messages=self.wire=self.commands=self.output=0
        self.peak_rss=self.peak_processes=0;self.last_sample=0
        self.pid=self.pidfd=self.identity=None;self.pipes=[];self.selector=selectors.DefaultSelector()
        self.channel=None;self.endpoint_stamp=None
        self.process_token=None;self.channel_token=None;self.process_epoch=0
        self.launch_state='NOT_ATTEMPTED';self.spawn_time=None;self.owned_time=None
        self.last_identity=None;self.signals=[];self.cleanup='NOT_STARTED'
        self.buffers=[0,0];self.status=None;self.reaped=False;self.ready=False;self.attached=[]
        self.mode_mutation=False
        self.sequence=0;self.input=bytearray();self.expected=None;self.foreground_launched=False
        self.last_mutation=None
        self.store=artifacts.ResourceStore(directory,fd,stream,self.monitor,self.event)
        self.spawns=[];self.closed=False
        self.event('OPENED',supervisor=os.getpid(),parent=parent)

    def event(self,phase,**facts):
        if self.events>=512:
            self.durable=False;raise engine.Failure('JournalPersistence')
        raw=json.dumps(dict(sequence=self.events+1,phase=phase,monotonic=time.monotonic(),**facts),
                       sort_keys=True,separators=(',',':')).encode()+b'\n'
        if self.journal_bytes+len(raw)>1048576:
            self.durable=False;raise engine.Failure('JournalPersistence')
        try:
            info=os.fstat(self.journal)
            path=os.stat('session.jsonl',dir_fd=self.fd,follow_symlinks=False)
            if info.st_nlink!=1 or engine._stamp(path)[:3]!=self.journal_stamp:
                raise engine.Failure('JournalPersistence')
            offset=0
            while offset<len(raw):
                n=os.write(self.journal,raw[offset:])
                if n<=0:raise engine.Failure('JournalPersistence')
                offset+=n
            os.fsync(self.journal)
        except BaseException as error:
            self.durable=False
            raise engine.Failure('JournalPersistence') from error
        self.events+=1;self.journal_bytes+=len(raw)

    def partial(self):
        return dict(first_fault=self.first_fault,launch_state=self.launch_state,known_pid=self.pid,
                    last_identity=self.last_identity,signals_issued=list(self.signals),cleanup=self.cleanup,
                    unusable_resources=[k for k,v in self.store.records.items() if v['state']=='UNUSABLE'],
                    journal_durable=self.durable)

    def fault(self,code):
        code={'unknown-identity':'IdentityUnknown','identity-changed':'IdentityChanged',
              'source-changed':'SourceChanged','absolute-path-required':'PathRetarget',
              'symlink':'AliasRefused','ownership-unproved':'LaunchUnknown',
              'cleanup-deadline':'CleanupUnknown'}.get(code,code)
        self.first_fault=self.first_fault or code
        if self.pid is not None and self.process_token is None and not self.reaped:
            self.launch_state='UNKNOWN'
        self.state='FAULTED';self.store.unusable()
        try:self.event('FAULTED',code=self.first_fault,launch_state=self.launch_state,pid=self.pid,
                       mutation=self.last_mutation)
        except BaseException:pass
        self.contain()

    def contain(self):
        if self.pid is None or self.reaped:
            self.cleanup='STOPPED' if self.launch_state=='NOT_ATTEMPTED' or self.reaped else 'UNKNOWN'
            self.state='STOPPED_UNUSABLE' if self.cleanup=='STOPPED' else 'CLEANUP_UNKNOWN'
            return
        self.state='CONTAINING'
        try:
            # The unreaped causal direct child is the positive signal anchor.
            engine.contain_owned(self.pid,self.signals)
            until=time.monotonic()+1
            while time.monotonic()<until:
                pid,status=os.waitpid(self.pid,os.WNOHANG)
                if pid:
                    self.status=os.waitstatus_to_exitcode(status);self.reaped=True;break
                time.sleep(.01)
            if not self.reaped:raise engine.Failure('CleanupUnknown')
            self.cleanup='STOPPED';self.state='STOPPED_UNUSABLE'
        except BaseException:
            self.cleanup='UNKNOWN';self.state='CLEANUP_UNKNOWN'

    def _consume(self,index,raw):
        self.buffers[index]+=len(raw);self.output+=len(raw)
        if self.buffers[index]>self.policy.stream_bytes or self.output>self.policy.output_bytes:
            raise engine.Failure('OutputLimit')

    def monitor(self):
        now=time.monotonic();self.polls+=1
        if self.polls>self.policy.polls or now-self.started>=self.policy.session_seconds:
            raise engine.Failure('LifetimeDeadline')
        if select.select([self.parent_fd],[],[],0)[0]:
            raise engine.Failure('ParentLost')
        poll=select.poll();poll.register(self.read,select.POLLHUP|select.POLLERR)
        if any(flags&(select.POLLHUP|select.POLLERR) for _,flags in poll.poll(0)):
            raise engine.Failure('ControlLost')
        if engine._current_stamp(self.directory)[:3]!=self.root_stamp:
            raise engine.Failure('PathRetarget')
        _source_check(False)
        if len(os.listdir('/proc/self/fd'))>64:
            raise engine.Failure('OutputLimit')
        for row in self.store.records.values():
            if row['kind'] in ('source','executable'):
                if engine._current_stamp(row['path'])!=tuple(row['stamp']):
                    raise engine.Failure('SourceChanged')
        sessions=()
        if self.pid is not None and not self.reaped:
            engine.drain_streams(self.selector,self._consume)
            if self.spawn_time is not None and now-self.spawn_time>=self.policy.total_seconds:
                raise engine.Failure('LifetimeDeadline')
            status=os.waitid(os.P_PID,self.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
            if status is not None:
                self.state='EXIT_OBSERVED'
                if status.si_code!=os.CLD_EXITED or status.si_status!=0:
                    raise engine.Failure('MutationUnusable' if self.mode_mutation
                                         else 'IdentityUnknown')
            elif self.identity is not None:
                try:
                    self.identity.verify()
                except engine.Failure as error:
                    if error.code!='unknown-identity':
                        raise
                    # Kernel exit can transiently remove exe before WNOWAIT reports.
                    # No IO/READY action uses this uncertainty: current() still refuses.
                    again=os.waitid(os.P_PID,self.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
                    if again is not None:
                        if again.si_code!=os.CLD_EXITED or again.si_status!=0:
                            raise
                        self.state='EXIT_OBSERVED'
            sessions=(self.pid,)
        if now-self.last_sample<.01:
            return
        _source_check()
        self.store.workspace_usage()
        try:
            count,rss=engine.sample_processes(sessions,(self.parent,os.getpid()))
        except engine.Failure:
            if select.select([self.parent_fd],[],[],0)[0]:
                raise engine.Failure('ParentLost')
            raise
        self.last_sample=time.monotonic()
        self.peak_processes=max(self.peak_processes,count);self.peak_rss=max(self.peak_rss,rss)
        if count>self.policy.processes:raise engine.Failure('ProcessLimit')
        if rss>self.policy.rss_bytes:raise engine.Failure('RSSLimit')

    def _io(self,fd,raw=None,maximum=None,deadline=None,complete=True):
        offset=0;result=bytearray()
        while True:
            self.monitor()
            if self.identity is not None:self.current()
            if time.monotonic()>=deadline:raise engine.Failure('ReadinessDeadline')
            reads=[fd] if raw is None else [];writes=[fd] if raw is not None else []
            readable,writable,_=select.select(reads,writes,[],min(.01,deadline-time.monotonic()))
            self.monitor()
            if self.identity is not None:self.current()
            if writable:
                try:n=os.write(fd,raw[offset:offset+16384])
                except BlockingIOError:continue
                if n<=0:raise engine.Failure('ControlLost')
                offset+=n
                self.monitor();self.current()
                if offset==len(raw):return offset
            if readable:
                try:chunk=os.read(fd,min(16384,maximum-len(result)))
                except BlockingIOError:continue
                self.monitor();self.current()
                if not chunk:raise engine.Failure('IdentityUnknown')
                result.extend(chunk)
                if not complete or len(result)>=maximum:return bytes(result)

    def _wait(self,deadline):
        self.monitor()
        if time.monotonic()>=deadline:raise engine.Failure('LifetimeDeadline')
        time.sleep(min(.01,deadline-time.monotonic()))
        self.monitor()

    def current(self):
        if self.identity is None or self.pid is None or self.reaped:
            raise engine.Failure('IdentityUnknown')
        self.identity.verify()
        if self.state=='EXIT_OBSERVED':
            raise engine.Failure('IdentityUnknown')
        # Source identity plus live kernel identity are both mandatory.
        if engine._current_stamp(self.command.tool.path)!=self.command.tool.identity:
            raise engine.Failure('SourceChanged')
        credentials=Path('/proc',str(self.pid),'status').read_text()
        uid=next(x for x in credentials.splitlines() if x.startswith('Uid:')).split()[2]
        gid=next(x for x in credentials.splitlines() if x.startswith('Gid:')).split()[2]
        obs=dict(pid=self.pid,startticks=self.identity.ticks,exe_devino=self.identity.executable,
                 exe_hash=self.command.tool.sha256,argv=self.command.argv,session_id=self.generation,
                 owner_epoch=self.process_epoch,observed_monotonic=time.monotonic(),
                 uid=int(uid),gid=int(gid))
        self.last_identity=obs
        return obs

    def command_spec(self,args):
        tool=self.store.get(args['tool'],('executable',))
        self.store.verify(tool)
        if not tool['stamp'][2]&0o111 or tool['stamp'][2]&(stat.S_ISUID|stat.S_ISGID):
            raise engine.Failure('AliasRefused')
        argv=[]
        for arg in args['argv']:
            if type(arg) is str:
                value=arg
            elif type(arg) is dict and set(arg)=={'prefix','ref','suffix'}:
                row=self.store.get(arg['ref'],('source','working'))
                self.store.verify(row)
                value=arg['prefix']+row['path']+arg['suffix']
            else:raise engine.Failure('InvalidLease')
            if '\0' in value:raise engine.Failure('InvalidLease')
            argv.append(value)
        env=args['env']
        if (not 1<=len(argv)<=128 or argv[0]!=tool['path']
                or sum(len(x.encode())+1 for x in argv)>16384
                or type(env) is not list or len(env)>64):
            raise engine.Failure('InvalidLease')
        environment={}
        for pair in env:
            if type(pair) is not list or len(pair)!=2:
                raise engine.Failure('InvalidLease')
            key,value=pair
            if (type(key) is not str or type(value) is not str or not key or '=' in key
                    or '\0' in key+value or key in environment):
                raise engine.Failure('InvalidLease')
            environment[key]=value
        if sum(len(k.encode())+len(v.encode())+2 for k,v in environment.items())>16384:
            raise engine.Failure('InvalidLease')
        self.store.verify_sources();self.store.workspace_check()
        return types.SimpleNamespace(tool=types.SimpleNamespace(path=tool['path'],identity=tuple(tool['stamp']),
                                                                sha256=tool['sha256']),
                                     argv=tuple(argv),env=tuple(environment.items()),cwd=str(self.directory),
                                     cwd_identity=self.root_stamp,input_bytes=b'')

    def spawn(self,args,mutation=False):
        if self.pid is not None and not self.reaped:raise engine.Failure('InvalidLease')
        if self.process_token is not None:raise engine.Failure('InvalidLease')
        for pair in self.pipes:
            for old_fd in pair:
                if old_fd>=0:
                    try:os.close(old_fd)
                    except OSError:pass
        self.pipes=[]
        if self.pidfd is not None:
            os.close(self.pidfd);self.pidfd=None
        self.identity=None;self.status=None;self.reaped=False
        self.buffers=[0,0];self.attached=[] if mutation else self.attached
        self.mode_mutation=mutation
        self.command=args.pop('_command',None) or self.command_spec(args)
        self.commands+=1
        if self.commands>self.policy.commands:raise engine.Failure('ProcessLimit')
        self.event('SPAWN_INTENT',argv=self.command.argv,mutation=mutation)
        self.state='SPAWN_INTENT';self.spawn_time=time.monotonic();self.launch_state='POSSIBLE_UNKNOWN'
        self.pid,self.pipes,witness=engine.spawn_exec(self.command,268435456)
        self.launch_state='SPAWNED_UNBOUND';self.state='SPAWNED_UNBOUND';self.pidfd=os.pidfd_open(self.pid)
        for index,(fd,_) in enumerate(self.pipes):
            self.selector.register(fd,selectors.EVENT_READ,index)
        self.spawns.append(dict(pid=self.pid,launch_state=self.launch_state,spawn_intent=self.spawn_time))
        self.event('SPAWNED_UNBOUND',pid=self.pid)
        deadline=min(self.spawn_time+self.policy.startup_seconds,self.started+self.policy.session_seconds,
                     self.spawn_time+self.policy.total_seconds,
                     self.spawn_time+self.policy.command_seconds if mutation else float('inf'))
        try:
            witnessed=False
            while True:
                self.monitor()
                if select.select([witness],[],[],0)[0]:
                    data=os.read(witness,64)
                    if data:raise engine.Failure('LaunchUnknown')
                    witnessed=True
                if self.state=='EXIT_OBSERVED':raise engine.Failure('IdentityUnknown')
                try:identity=engine.Identity.observe(self.pid)
                except engine.Failure:identity=None
                if (witnessed and identity is not None and identity.matches(self.command)
                        and identity.session==self.pid):
                    self.identity=identity;self.owned_time=time.monotonic();self.launch_state='CURRENTLY_OWNED'
                    self.state='OWNED';self.process_epoch+=1
                    observation=self.current()
                    self.event('OWNED',pid=self.pid,startticks=identity.ticks)
                    self.spawns[-1].update(launch_state=self.launch_state,startticks=identity.ticks)
                    return observation
                if time.monotonic()>=deadline:raise engine.Failure('StartupDeadline')
                self._wait(deadline)
        finally:
            os.close(witness)

    def process_ref(self,ref):
        if (type(ref) is not dict or ref!={'token':self.process_token,'epoch':self.process_epoch}
                or self.process_token is None):
            raise engine.Failure('InvalidLease')
        self.current()

    def channel_ref(self,ref):
        if (type(ref) is not dict or ref!={'token':self.channel_token,'epoch':self.process_epoch}
                or self.channel_token is None or self.channel is None):
            raise engine.Failure('InvalidLease')
        self.current()
        if engine._stamp(os.stat(self.endpoint_path,follow_symlinks=False))!=self.endpoint_stamp:
            raise engine.Failure('PathRetarget')
        pid,uid,gid=struct.unpack('3i',self.channel.getsockopt(socket.SOL_SOCKET,socket.SO_PEERCRED,12))
        obs=self.current()
        if False:
            raise engine.Failure('ForeignPeer')

    def endpoint(self,args):
        self.process_ref(args['process'])
        name=args['basename']
        if type(name) is not str or name in ('','.','..') or '/' in name or '\0' in name or len(name)>128:
            raise engine.Failure('PathRetarget')
        path=engine._path(self.directory/name)
        self.event('ENDPOINT_INTENT',path=str(path))
        deadline=min(self.owned_time+self.policy.readiness_seconds,
                     self.spawn_time+self.policy.total_seconds,self.started+self.policy.session_seconds)
        while True:
            self.monitor();self.current()
            try:info=path.stat(follow_symlinks=False)
            except FileNotFoundError:
                self._wait(deadline);continue
            if not stat.S_ISSOCK(info.st_mode):raise engine.Failure('ForeignPeer')
            self.endpoint_stamp=engine._stamp(info);self.event('ENDPOINT_PRESENT');break
        connection=socket.socket(socket.AF_UNIX);connection.setblocking(False)
        status=connection.connect_ex(str(path))
        if status not in (0,errno.EINPROGRESS,errno.EAGAIN,errno.EALREADY):
            connection.close();raise engine.Failure('ForeignPeer')
        while True:
            self.monitor();self.current()
            if time.monotonic()>=deadline:
                connection.close();raise engine.Failure('ReadinessDeadline')
            if select.select([],[connection],[],.01)[1]:break
        if connection.getsockopt(socket.SOL_SOCKET,socket.SO_ERROR):
            connection.close();raise engine.Failure('ForeignPeer')
        self.event('ENDPOINT_CONNECTED')
        self.channel=connection;self.endpoint_path=path;self.channel_token=uuid.uuid4().hex
        ref=dict(token=self.channel_token,epoch=self.process_epoch,kind='channel')
        self.channel_ref(dict(token=self.channel_token,epoch=self.process_epoch))
        self.store.verify_sources()
        self.event('ENDPOINT_VERIFIED')
        return ref

    def channel_io(self,operation,args):
        self.channel_ref(args['channel'])
        timeout=args['timeout']
        if type(timeout) not in (int,float) or not math.isfinite(timeout) or not 0<timeout<=60:
            raise engine.Failure('ReadinessDeadline')
        deadline=min(time.monotonic()+timeout,self.spawn_time+self.policy.total_seconds,
                     self.started+self.policy.session_seconds)
        tx=rx=0
        if operation=='exchange':
            if self.ready or self.state!='OWNED':raise engine.Failure('InvalidLease')
            deadline=min(deadline,self.owned_time+self.policy.readiness_seconds)
            request=bytes.fromhex(args['request']);expected=bytes.fromhex(args['reply'])
            if not 0<len(request)<=4096 or not 0<len(expected)<=4096:
                raise engine.Failure('OutputLimit')
            tx=self._io(self.channel.fileno(),raw=request,deadline=deadline)
            reply=self._io(self.channel.fileno(),maximum=len(expected),deadline=deadline)
            rx=len(reply)
            if reply!=expected:raise engine.Failure('IdentityChanged')
            self.channel_ref(args['channel']);self.store.verify_sources()
            observation=self.current()
            self.monitor();self.channel_ref(args['channel'])
            self.ready=True;self.state='READY';self.event('READY',pid=self.pid)
            return dict(current_identity=observation,request_sha=hashlib.sha256(request).hexdigest(),
                        reply_sha=hashlib.sha256(reply).hexdigest(),bytes_tx=tx,bytes_rx=rx,
                        observed_monotonic=time.monotonic(),ready=True,source_epoch=self.store.source_epoch)
        if self.state!='READY':raise engine.Failure('InvalidLease')
        if operation=='send':
            raw=bytes.fromhex(args['raw'])
            if not 0<len(raw)<=4096:raise engine.Failure('OutputLimit')
            tx=self._io(self.channel.fileno(),raw=raw,deadline=deadline)
            self.channel_ref(args['channel'])
            return dict(bytes_tx=tx,bytes_rx=0,observed_monotonic=time.monotonic())
        maximum=args['max_bytes']
        if type(maximum) is not int or not 0<maximum<=4096:raise engine.Failure('OutputLimit')
        raw=self._io(self.channel.fileno(),maximum=maximum,deadline=deadline,complete=False)
        self.channel_ref(args['channel']);return raw.hex()

    def passive_finish(self,timeout):
        if type(timeout) not in (int,float) or not math.isfinite(timeout) or not 0<timeout<=10:
            raise engine.Failure('LifetimeDeadline')
        deadline=min(time.monotonic()+timeout,self.spawn_time+self.policy.total_seconds,
                     self.started+self.policy.session_seconds,
                     self.spawn_time+self.policy.command_seconds if self.mode_mutation else float('inf'))
        while True:
            self.monitor()
            status=os.waitid(os.P_PID,self.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
            if status is not None:
                if status.si_code!=os.CLD_EXITED or status.si_status!=0:
                    raise engine.Failure('MutationUnusable')
                if any(p!=self.pid and state!='Z' for p,state,_ in engine._session_members(self.pid)):
                    raise engine.Failure('CleanupUnknown')
                engine.drain_streams(self.selector,self._consume)
                if not self.selector.get_map():break
            if time.monotonic()>=deadline:raise engine.Failure('LifetimeDeadline')
            self._wait(deadline)
        _,status=os.waitpid(self.pid,0);self.status=os.waitstatus_to_exitcode(status);self.reaped=True
        self.cleanup='STOPPED';self.state='STOPPED'
        for token,_ in self.attached:self.store.records[token]['state']='DIRTY_STOPPED'
        self.event('STOPPED',pid=self.pid,exit_code=self.status,signals=[])
        return dict(created_pid=self.pid,startticks=self.identity.ticks,last_verified_identity=self.last_identity,
                    exit_code=self.status,reaped=True,signals_issued=[],owned_session_remaining=[],
                    identity_at_exit_unproved=True,daemon_state='UNKNOWN')

    def handle(self,operation,args):
        if operation in ('capture','prepare','tool','snapshot','mutate'):
            self.store.operation_deadline=time.monotonic()+self.stream.operation_seconds
        try:
            return self.dispatch(operation,args)
        finally:
            self.store.operation_deadline=None

    def dispatch(self,operation,args):
        if operation=='close':
            if self.pid is not None and not self.reaped:
                self.fault('Cancelled')
            self.closed=True
            return self.outcome()
        if operation=='fault':
            raise engine.Failure(args['code'])
        if operation=='cancel':
            raise engine.Failure('Cancelled')
        if self.first_fault:
            raise engine.Failure(self.first_fault)
        self.monitor();_source_check()
        if operation=='capture':return self.store.capture(args)
        if operation=='prepare':return self.store.prepare(args['source'],args['basename'])
        if operation=='tool':return self.store.tool(args['source'])
        if operation=='snapshot':return self.store.snapshot(args['target'])
        if operation=='launch':
            if self.foreground_launched or (self.pid is not None and not self.reaped):raise engine.Failure('InvalidLease')
            self.foreground_launched=True
            args['_command']=self.command_spec(args)
            rows=[self.store.get(x,('working',)) for x in args['resources']]
            if len({id(x) for x in rows})!=len(rows):raise engine.Failure('InvalidLease')
            for ref,row in zip(args['resources'],rows):
                self.store.verify(row);row['state']='BUSY';self.attached.append((ref['token'],'BUSY'))
            self.store.launched=True
            obs=self.spawn(args);self.process_token=uuid.uuid4().hex
            return dict(token=self.process_token,epoch=self.process_epoch,kind='process',observation=obs)
        if operation=='identity':
            self.process_ref(args['process']);return self.current()
        if operation=='endpoint':return self.endpoint(args)
        if operation in ('exchange','send','receive'):return self.channel_io(operation,args)
        if operation=='channel_close':
            self.channel_ref(args['channel']);self.channel.close();self.channel=None;self.channel_token=None
            return None
        if operation=='finish':
            # Passive finish remains legal after an observed direct exit.
            ref=args['process']
            if ref!={'token':self.process_token,'epoch':self.process_epoch}:raise engine.Failure('InvalidLease')
            return self.passive_finish(args['timeout'])
        if operation=='mutate':
            return self.mutate(args)
        raise engine.Failure('InvalidLease')

    def mutate(self,args):
        if self.foreground_launched:
            raise engine.Failure('InvalidLease')
        row=self.store.get(args['target'],('working',))
        if row['state']!='READY':
            raise engine.Failure('InvalidLease')
        before_epoch=row['epoch']
        for ref in args['inputs']:
            self.store.get(ref,('source',))
        allowed_sources={ref['token'] for ref in args['inputs']}
        for value in args['argv']:
            if type(value) is dict:
                ref=value['ref']
                record=self.store.get(ref,('source','working'))
                if record['kind']=='working' and ref!=args['target']:
                    raise engine.Failure('InvalidLease')
                if record['kind']=='source' and ref['token'] not in allowed_sources:
                    raise engine.Failure('InvalidLease')
        command=self.command_spec(args)
        self.store.verify(row)
        immutable_working=[r for r in self.store.records.values()
                           if r['kind']=='working' and r is not row]
        baseline={id(r):tuple(r['stamp']) for r in immutable_working}
        self.event('MUTATION_INTENT',resource=args['target']['token'],epoch=before_epoch)
        row['state']='MUTATING'
        try:
            args['_command']=command
            observed=self.spawn(args,mutation=True)
            self.attached=[(args['target']['token'],'MUTATING')]
            result=self.passive_finish(min(10,self.policy.command_seconds))
            self.store.verify_sources()
            for other in immutable_working:
                if tuple(other['stamp'])!=baseline[id(other)]:
                    raise engine.Failure('SourceChanged')
                self.store.verify(other)
            stamp,size,digest=self.store.stream(row['path'])
            if (stamp[:3]!=tuple(row['stamp'][:3]) or stamp[6]!=1
                    or engine._stamp(os.fstat(row['fd']))!=stamp):
                raise engine.Failure('PathRetarget')
            self.store.workspace_check()
            self.event('MUTATION_COMMITTED',resource=args['target']['token'],
                       before_epoch=before_epoch,after_epoch=before_epoch+1,size=size,sha256=digest)
            row.update(stamp=stamp,size=size,sha256=digest,epoch=before_epoch+1,state='READY')
            self.attached=[]
            self.last_mutation=dict(status='COMMITTED',before_epoch=before_epoch,after_epoch=row['epoch'],
                                    tool_identity=observed,actual_argv=command.argv,exit_code=result['exit_code'],
                                    partial_effects=True,immutable_inputs_unchanged=True,
                                    journal_sequence=self.events)
            return dict(resource=self.store._reference(row),result=self.last_mutation)
        except BaseException:
            row['state']='UNUSABLE'
            self.last_mutation=dict(status='UNUSABLE',before_epoch=before_epoch,after_epoch=None,
                                    tool_identity=self.last_identity,actual_argv=command.argv,
                                    exit_code=self.status,partial_effects=True,
                                    immutable_inputs_unchanged=False,journal_sequence=self.events,
                                    successor=None)
            raise

    def outcome(self):
        return dict(terminal_state='CLOSED',first_fault=self.first_fault,charged_operations=self.commands,
                    spawn_observations=self.spawns,
                    resources=[self.store.describe(x) for x in self.store.records.values()],
                    counters=[['source_io',self.store.io],['prepared_bytes',self.store.copied],
                              ['max_chunk',self.store.max_chunk],['peak_rss',self.peak_rss],
                              ['peak_processes',self.peak_processes],['polls',self.polls],
                              ['output_bytes',self.output],['journal_events',self.events],
                              ['journal_bytes',self.journal_bytes]],
                    elapsed=time.monotonic()-self.started,cleanup=self.cleanup)

    def reply(self,value,ok=True,error=None):
        response=dict(generation=self.generation,sequence=self.sequence,ok=ok)
        if ok:response['value']=value
        else:response.update(error=error,partial=self.partial(),
                             mutation_result=self.last_mutation if self.last_mutation and self.last_mutation['status']=='UNUSABLE' else None)
        raw=_frame(json.dumps(response,separators=(',',':'),allow_nan=False).encode())
        offset=0;deadline=time.monotonic()+1
        while offset<len(raw):
            if time.monotonic()>=deadline:raise engine.Failure('ControlLost')
            if select.select([self.parent_fd],[],[],0)[0]:raise engine.Failure('ParentLost')
            if select.select([],[self.write],[],.01)[1]:
                try:n=os.write(self.write,raw[offset:offset+16384])
                except BlockingIOError:continue
                if n<=0:raise engine.Failure('ControlLost')
                offset+=n
        self.wire+=len(raw)
        if self.wire>67108864:raise engine.Failure('ControlLost')

    def loop(self):
        try:
            self.reply(dict(supervisor=os.getpid(),parent=self.parent,root=self.root_stamp,
                            generation=self.generation,source=sys.modules[__package__].source_binding(),
                            journal_identity=self.journal_stamp))
            while not self.closed:
                try:self.monitor()
                except BaseException as error:
                    if not self.first_fault:self.fault(error.code if isinstance(error,engine.Failure) else 'ControlLost')
                    # Dead parent/channel requires independent containment, then exit.
                    if self.first_fault in ('ParentLost','ControlLost'):break
                readable=select.select([self.read],[],[],.01)[0]
                if not readable:continue
                chunk=os.read(self.read,16384)
                if not chunk:
                    self.fault('ControlLost');break
                self.input.extend(chunk);self.wire+=len(chunk)
                if len(self.input)>65540 or self.wire>67108864:
                    self.fault('ControlLost');break
                if len(self.input)<4:continue
                size=struct.unpack('!I',self.input[:4])[0]
                if size>65536:
                    self.fault('ControlLost');break
                if len(self.input)<size+4:continue
                if len(self.input)!=size+4:
                    self.fault('ControlLost');break
                raw=bytes(self.input[4:]);self.input.clear()
                try:
                    request=_json(raw);self.messages+=1
                    if (self.messages>4096 or request['generation']!=self.generation
                            or request['sequence']!=self.sequence+1):
                        raise engine.Failure('InvalidLease')
                    self.sequence=request['sequence']
                    value=self.handle(request['operation'],request['args'])
                    self.reply(value)
                except BaseException as error:
                    code=error.code if isinstance(error,engine.Failure) else 'ControlLost'
                    self.fault(code);self.reply(None,False,self.first_fault)
        finally:
            if self.pid is not None and not self.reaped:self.fault(self.first_fault or 'ControlLost')
            if self.channel is not None:self.channel.close()
            for pair in self.pipes:
                for fd in pair:
                    if fd>=0:
                        try:os.close(fd)
                        except OSError:pass
            for fd in self.store.resource_descriptors:
                try:os.close(fd)
                except OSError:pass
            for fd in (self.pidfd,self.parent_fd,self.read,self.write,self.journal,self.fd):
                if fd is not None and fd>=0:
                    try:os.close(fd)
                    except OSError:pass
            self.selector.close()

    if (type(workspace) is not artifacts.NewWorkspace or type(workspace.path) is not str
            or type(stream) is not artifacts.StreamPolicy
            or type(managed) is not ManagedPolicy or threading.active_count()!=1
            or threading.current_thread() is not threading.main_thread()):
        raise StartupFailure('InvalidLease',_partial(None))
    _source_check()

    fd=journal=parent_fd=-1;pipes=[];pid=None;pidfd=None
    read=write=-1;durable=False
    generation=uuid.uuid4().hex
    try:
        os.mkdir(directory,0o700)
        fd=engine._open_path(directory,os.O_RDONLY|os.O_DIRECTORY)
        workspace_stamp=engine._stamp(os.fstat(fd))[:3]
        journal=os.open('session.jsonl',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600,dir_fd=fd)
        journal_stamp=engine._stamp(os.fstat(journal))[:3]

                for entry in os.listdir('/proc/self/fd'):
                    item=int(entry)
                    if item not in keep:
                        try:os.close(item)
                        except OSError:pass
                resource.setrlimit(resource.RLIMIT_AS,(268435456,268435456))
                resource.setrlimit(resource.RLIMIT_CORE,(0,0))
                resource.setrlimit(resource.RLIMIT_NOFILE,(64,64))
                worker=_Worker(directory,fd,journal,read,write,parent,parent_fd,generation,stream,managed)
                worker.loop();os._exit(0)
            except BaseException:
                os._exit(125)
        os.close(parent_fd);parent_fd=-1;os.close(fd);fd=-1;os.close(journal);journal=-1
        os.close(pipes[0][0]);os.close(pipes[1][1])
        write,read=pipes[0][1],pipes[1][0];pipes=[]

        session=Session(_SESSION_KEY);session._initialize(pid,pidfd,read,write,generation,managed,str(directory))

                raise engine.Failure('LaunchUnknown')
            if not select.select([read],[],[],.01)[0]:continue
            chunk=os.read(read,16384)
            if not chunk:raise engine.Failure('ControlLost')
            received.extend(chunk)
            if len(received)>65540:raise engine.Failure('ControlLost')
            if len(received)<4:continue
            size=struct.unpack('!I',received[:4])[0]
            if size>65536:raise engine.Failure('ControlLost')
            if len(received)<size+4:continue
            if len(received)!=size+4:raise engine.Failure('ControlLost')

            if (hello.get('generation')!=generation or hello.get('sequence')!=0 or not hello.get('ok')
                    or hello['value']['supervisor']!=pid or hello['value']['parent']!=parent
                    or tuple(hello['value']['root'])!=workspace_stamp

            fields=Path('/proc',str(pid),'stat').read_text().rsplit(')',1)[1].split()
            if int(fields[1])!=parent:raise engine.Failure('LaunchUnknown')
            observed.verify();break
        _ACTIVE_SESSION=session
        return session
    except BaseException as error:
        for pair in pipes:
            for item in pair:
                try:os.close(item)
                except OSError:pass
        for item in (fd,journal,parent_fd):
            if item>=0:
                try:os.close(item)
                except OSError:pass
        if pid is not None:
            # Only close owned control ends. Worker independently detects parent/control loss.
            for item in (locals().get('write',-1),locals().get('read',-1)):
                if item>=0:
                    try:os.close(item)
                    except OSError:pass
        code=error.code if isinstance(error,engine.Failure) else ('NoClobber' if isinstance(error,FileExistsError) else 'ControlLost')
        reaped=False

            while time.monotonic()<until:
                try:done,_=os.waitpid(pid,os.WNOHANG)
                except ChildProcessError:done=pid
                if done:
                    reaped=True;break
                time.sleep(.01)
        if pidfd is not None:
            os.close(pidfd)
        observation=PartialOutcome(code,'NOT_ATTEMPTED',None,None,(),
                                   'STOPPED' if pid is None or reaped else 'UNKNOWN',(),durable)
        raise StartupFailure(code,observation) from error
